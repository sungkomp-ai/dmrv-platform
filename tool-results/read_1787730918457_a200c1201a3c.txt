    70→interface BufferEntry {
    71→  id: string;
    72→  projectId: string;
    73→  creditId: string;
    74→  amount: number;
    75→  uncertaintyPct: number;
    76→  bufferPct: number;
    77→  riskCategory: 'LOW' | 'MEDIUM' | 'HIGH' | 'VERY_HIGH';
    78→  reason: string;
    79→  releasedAt: string | null;
    80→  createdAt: string;
    81→  project: {
    82→    id: string;
    83→    name: string;
    84→    province: string;
    85→  };
    86→  credit: {
    87→    id: string;
    88→    tokenId: string;
    89→    amount: number;
    90→    status: string;
    91→  };
    92→}
    93→
    94→interface ByRiskCategory {
    95→  LOW: number;
    96→  MEDIUM: number;
    97→  HIGH: number;
    98→  VERY_HIGH: number;
    99→}
   100→
   101→interface ByProjectEntry {
   102→  projectName: string;
   103→  province: string;
   104→  amount: number;
   105→  count: number;
   106→}
   107→
   108→interface BufferSummary {
   109→  total: number;
   110→  totalBuffered: number;
   111→  byRiskCategory: ByRiskCategory;
   112→  byProject: Record<string, ByProjectEntry>;
   113→}
   114→
   115→interface BufferResponse {
   116→  entries: BufferEntry[];
   117→  summary: BufferSummary;
   118→}

   231→export default function BufferPoolView() {
   232→  const [data, setData] = useState<BufferResponse | null>(null);
   233→  const [loading, setLoading] = useState(true);
   234→  const [error, setError] = useState<string | null>(null);
   235→  const [refreshing, setRefreshing] = useState(false);
   236→  const [releaseDialogOpen, setReleaseDialogOpen] = useState(false);
   237→  const [selectedEntryId, setSelectedEntryId] = useState<string>('');
   238→  const [releaseReason, setReleaseReason] = useState('');
   239→  const [submitting, setSubmitting] = useState(false);
   240→
   241→  // -----------------------------------------------------------------------
   242→  // Data fetching
   243→  // -----------------------------------------------------------------------
   244→
   245→  const fetchData = useCallback(async () => {
   246→    try {
   247→      setError(null);
   248→      const res = await fetch('/api/dmrv/buffer');
   249→      if (!res.ok) throw new Error(`Failed to fetch buffer data: ${res.status}`);
   250→      const json: BufferResponse = await res.json();
   251→      setData(json);
   252→    } catch (err) {
   253→      setError(err instanceof Error ? err.message : 'Unknown error');
   254→    } finally {
   255→      setLoading(false);
   256→      setRefreshing(false);
   257→    }
   258→  }, []);
   259→
   260→  useEffect(() => {
   261→    setLoading(true);
   262→    fetchData();
   263→  }, [fetchData]);
   264→
   265→  const handleRefresh = () => {
   266→    setRefreshing(true);
   267→    fetchData();
   268→  };
   269→
   270→  // -----------------------------------------------------------------------
   271→  // Derived data
   272→  // -----------------------------------------------------------------------
   273→
   274→  const entries = data?.entries ?? [];
   275→  const summary = data?.summary;
   276→
   277→  const totalBuffered = summary?.totalBuffered ?? 0;
   278→  const totalCredits = entries.reduce((sum, e) => sum + e.credit.amount, 0);
   279→  const bufferPoolPct = totalCredits > 0 ? (totalBuffered / totalCredits) * 100 : 0;
   280→  const avgUncertainty =
   281→    entries.length > 0
   282→      ? entries.reduce((sum, e) => sum + e.uncertaintyPct, 0) / entries.length
   283→      : 0;
   284→  const entriesCount = summary?.total ?? entries.length;
   285→
   286→  const heldEntries = entries.filter((e) => !e.releasedAt);
   287→
   288→  // Pie chart data
   289→  const riskCategoryData = summary
   290→    ? Object.entries(summary.byRiskCategory).map(([key, value]) => ({
   291→        name: key === 'VERY_HIGH' ? 'VERY HIGH' : key,
   292→        value,
   293→        color: RISK_COLORS[key] || '#94a3b8',
   294→      }))
   295→    : [];
   296→
   297→  // Bar chart data
   298→  const projectBarData = summary
   299→    ? Object.entries(summary.byProject).map(([, proj]) => {
   300→        const projectEntries = entries.filter((e) => e.project.name === proj.projectName);
   301→        const buffered = projectEntries.reduce((s, e) => s + e.amount, 0);
   302→        const creditTotal = projectEntries.reduce((s, e) => s + e.credit.amount, 0);
   303→        const available = creditTotal - buffered;
   304→        return {
   305→          name: proj.projectName.length > 16 ? proj.projectName.substring(0, 14) + '...' : proj.projectName,
   306→          Available: Math.round(available * 100) / 100,
   307→          Buffered: Math.round(buffered * 100) / 100,
   308→        };
   309→      })
   310→    : [];
   311→
   312→  // -----------------------------------------------------------------------
   313→  // Release handler
   314→  // -----------------------------------------------------------------------
   315→
   316→  const handleRelease = async () => {
   317→    if (!selectedEntryId) {
   318→      toast.error('Please select an entry to release.');
   319→      return;
   320→    }
   321→    setSubmitting(true);
   322→    try {
   323→      const res = await fetch('/api/dmrv/buffer', {
   324→        method: 'POST',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({ action: 'release', entryId: selectedEntryId }),
   327→      });
   328→      if (!res.ok) {
   329→        const body = await res.json().catch(() => ({}));
   330→        throw new Error(body.error || `Release failed with status ${res.status}`);
   331→      }
   332→      toast.success('Buffer credits released successfully.');
   333→      setReleaseDialogOpen(false);
   334→      setSelectedEntryId('');
   335→      setReleaseReason('');
   336→      fetchData();
   337→    } catch (err) {
   338→      toast.error(err instanceof Error ? err.message : 'Failed to release buffer credits.');
   339→    } finally {
   340→      setSubmitting(false);
   341→    }
   342→  };
   343→
   344→  // -----------------------------------------------------------------------
   345→  // Render: Loading
   346→  // -----------------------------------------------------------------------
   347→
   348→  if (loading) {
   349→    return (
   350→      <div className="space-y-6">
   351→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→          <div>
   353→            <Skeleton className="h-8 w-64" />
   354→            <Skeleton className="h-4 w-80 mt-2" />
   355→          </div>
   356→          <Skeleton className="h-9 w-24" />
   357→        </div>
   358→        <div className="flex gap-4 overflow-x-auto pb-2">
   359→          {Array.from({ length: 5 }).map((_, i) => (
   360→            <StatCardSkeleton key={i} />
   361→          ))}
   362→        </div>
   363→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   364→          <ChartSkeleton />
   365→          <ChartSkeleton />
   366→        </div>
   367→        <TableSkeleton />
   368→      </div>
   369→    );
   370→  }
   371→
   372→  // -----------------------------------------------------------------------
   373→  // Render: Error
   374→  // -----------------------------------------------------------------------
   375→
   376→  if (error) {
   377→    return (
   378→      <div className="space-y-6">
   379→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   380→          <div>
   381→            <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   382→              <Landmark className="h-6 w-6 text-emerald-600" />
   383→              Uncertainty &amp; Buffer Pool
   384→            </h2>
   385→            <p className="text-muted-foreground text-sm mt-1">
   386→              Risk reserve management per Verra/Gold Standard
   387→            </p>
   388→          </div>
   389→        </div>
   390→        <Card className="border-red-200 dark:border-red-800">
   391→          <CardContent className="py-12 flex flex-col items-center gap-4">
   392→            <AlertTriangle className="h-10 w-10 text-red-500" />
   393→            <div className="text-center">
   394→              <p className="font-semibold text-red-700 dark:text-red-400">Failed to load buffer pool data</p>
   395→              <p className="text-sm text-muted-foreground mt-1">{error}</p>
   396→            </div>
   397→            <Button variant="outline" onClick={handleRefresh} className="gap-2">
   398→              <RefreshCw className="h-4 w-4" />
   399→              Retry
   400→            </Button>
   401→          </CardContent>
   402→        </Card>
   403→      </div>
   404→    );
   405→  }
   406→
   407→  // -----------------------------------------------------------------------
   408→  // Render: Main
   409→  // -----------------------------------------------------------------------
   410→
   411→  return (
   412→    <div className="space-y-6">
   413→      {/* ----------------------------------------------------------------- */}
   414→      {/* 1. Header                                                         */}
   415→      {/* ----------------------------------------------------------------- */}
   416→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   417→        <div>
   418→          <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   419→            <Landmark className="h-6 w-6 text-emerald-600" />
   420→            Uncertainty &amp; Buffer Pool
   421→          </h2>
   422→          <p className="text-muted-foreground text-sm mt-1">
   423→            Risk reserve management per Verra/Gold Standard
   424→          </p>
   425→        </div>
   426→        <Button variant="outline" size="sm" className="gap-2 self-start" onClick={handleRefresh} disabled={refreshing}>
   427→          <RefreshCw className={`h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />
   428→          Refresh
   429→        </Button>
   430→      </div>
   431→
   432→      {/* ----------------------------------------------------------------- */}
   433→      {/* 2. Summary Stat Cards                                             */}
   434→      {/* ----------------------------------------------------------------- */}
   435→      <ScrollArea className="w-full whitespace-nowrap">
   436→        <div className="flex gap-4 pb-2">
   437→          {/* Total Buffered */}
   438→          <Card className="min-w-[170px] flex-1">
   439→            <CardContent className="p-4 flex items-center gap-3">
   440→              <div className="h-10 w-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   441→                <ShieldCheck className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   442→              </div>
   443→              <div>
   444→                <p className="text-2xl font-bold">{totalBuffered.toLocaleString()}</p>
   445→                <p className="text-xs text-muted-foreground">Total Buffered (tCO2e)</p>
   446→              </div>
   447→            </CardContent>
   448→          </Card>
   449→
   450→          {/* Total Credits Generated */}
   451→          <Card className="min-w-[170px] flex-1">
   452→            <CardContent className="p-4 flex items-center gap-3">
   453→              <div className="h-10 w-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   454→                <TrendingUp className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   455→              </div>
   456→              <div>
   457→                <p className="text-2xl font-bold">{totalCredits.toLocaleString()}</p>

   459→              </div>
   460→            </CardContent>
   461→          </Card>
   462→
   463→          {/* Buffer Pool % */}
   464→          <Card className="min-w-[170px] flex-1">
   465→            <CardContent className="p-4 flex items-center gap-3">
   466→              <div className="h-10 w-10 rounded-lg bg-amber-100 dark:bg-amber-900/40 flex items-center justify-center">
   467→                <Percent className="h-5 w-5 text-amber-600 dark:text-amber-400" />
   468→              </div>
   469→              <div>
   470→                <p className="text-2xl font-bold">{bufferPoolPct.toFixed(1)}%</p>

   476→          {/* Avg Uncertainty % */}
   477→          <Card className="min-w-[170px] flex-1">
   478→            <CardContent className="p-4 flex items-center gap-3">
   479→              <div className="h-10 w-10 rounded-lg bg-orange-100 dark:bg-orange-900/40 flex items-center justify-center">

   489→          {/* Entries Count */}

   491→            <CardContent className="p-4 flex items-center gap-3">
   492→              <div className="h-10 w-10 rounded-lg bg-slate-100 dark:bg-slate-800 flex items-center justify-center">
   493→                <List className="h-5 w-5 text-slate-600 dark:text-slate-400" />
   494→              </div>
   495→              <div>
   496→                <p className="text-2xl font-bold">{entriesCount}</p>
   497→                <p className="text-xs text-muted-foreground">Entries Count</p>
   498→              </div>
   499→            </CardContent>
   500→          </Card>
   501→        </div>
   502→        <ScrollBar orientation="horizontal" />
   503→      </ScrollArea>
   504→
   505→      {/* ----------------------------------------------------------------- */}
   506→      {/* 3. Uncertainty Gauge                                              */}
   507→      {/* ----------------------------------------------------------------- */}
   508→      <Card>
   509→        <CardHeader className="pb-2">
   510→          <CardTitle className="text-base flex items-center gap-2">
   511→            <Info className="h-4 w-4 text-muted-foreground" />
   512→            System Uncertainty Level
   513→          </CardTitle>
   514→          <CardDescription>Average uncertainty across all buffer entries</CardDescription>
   515→        </CardHeader>
   516→        <CardContent className="pb-6">
   517→          <div className="space-y-3">
   518→            {/* Gradient bar */}
   519→            <div className="relative h-6 w-full rounded-full overflow-hidden">
   520→              <div
   521→                className="h-full w-full rounded-full"
   522→                style={{
   523→                  background: 'linear-gradient(to right, #22c55e, #84cc16, #f59e0b, #f97316, #ef4444)',
   524→                }}
   525→              />
   526→              {/* Pointer */}
   527→              <div
   528→                className="absolute top-0 h-full flex flex-col items-center"
   529→                style={{
   530→                  left: `${Math.min(Math.max(avgUncertainty, 0), 60) / 60 * 100}%`,
   531→                  transform: 'translateX(-50%)',
   532→                }}
   533→              >
   534→                <div className="w-0.5 h-full bg-white dark:bg-gray-900 shadow-sm" />
   535→              </div>
   536→            </div>
   537→            {/* Scale labels */}
   538→            <div className="flex justify-between text-xs text-muted-foreground px-0.5">
   539→              <span>0%</span>
   540→              <span>15%</span>
   541→              <span>30%</span>

   544→            </div>

   549→              </span>
   550→              <Badge
   551→                variant="outline"
   552→                className={
   553→                  avgUncertainty < 15
   554→                    ? 'border-green-300 text-green-700 dark:text-green-400 dark:border-green-700'
   555→                    : avgUncertainty < 30
   556→                      ? 'border-amber-300 text-amber-700 dark:text-amber-400 dark:border-amber-700'
   557→                      : avgUncertainty < 45
   558→                        ? 'border-orange-300 text-orange-700 dark:text-orange-400 dark:border-orange-700'
