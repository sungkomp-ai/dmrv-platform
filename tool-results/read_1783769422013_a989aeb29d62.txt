   193→   193→// ─── Helper: Generate Integrity Hash ──────────────────────────────────
   194→   194→function generateHash(sensorId: string, value: number, timestamp: string): string {
   195→   195→  // Simplified SHA-like hash for demo
   196→   196→  const raw = `${sensorId}:${value}:${timestamp}`;
   197→   197→  let hash = 0;
   198→   198→  for (let i = 0; i < raw.length; i++) {
   199→   199→    const char = raw.charCodeAt(i);
   200→   200→    hash = ((hash << 5) - hash) + char;
   201→   201→    hash = hash & hash;
   202→   202→  }
   203→   203→  return `sha256-${Math.abs(hash).toString(16).padStart(16, '0')}`;
   204→   204→}
   205→   205→
   206→   206→function generateSignature(nodeId: string, hash: string): string {
   207→   207→  const raw = `${nodeId}:${hash}`;
   208→   208→  let sig = 0;
   209→   209→  for (let i = 0; i < raw.length; i++) {
   210→   210→    const char = raw.charCodeAt(i);
   211→   211→    sig = ((sig << 5) - sig) + char;
   212→   212→    sig = sig & sig;
   213→   213→  }
   214→   214→  return `sig-${Math.abs(sig).toString(16).padStart(16, '0')}`;
   215→   215→}

   217→   217→function validateReading(sensor: SensorConfig, value: number): 'VALIDATED' | 'ANOMALY' | 'REJECTED' {
   218→   218→  if (value < sensor.minRange || value > sensor.maxRange) return 'REJECTED';
   219→   219→  // Anomaly detection: if value is in top/bottom 5% of range
   220→   220→  const range = sensor.maxRange - sensor.minRange;
   221→   221→  const lowerBound = sensor.minRange + range * 0.02;
   222→   222→  const upperBound = sensor.maxRange - range * 0.02;
   223→   223→  if (value < lowerBound || value > upperBound) return 'ANOMALY';
   224→   224→  return 'VALIDATED';
   225→   225→}

   245→   245→function generateSensorValue(sensorType: string, sensor: SensorConfig, tick: number): number {
   246→   246→  const config = BASE_VALUES[sensorType] || { base: 50, variance: 10 };
   247→   247→  const hour = (tick % 24);
   248→   248→  const seasonalFactor = config.seasonal ? Math.sin((hour / 24) * Math.PI * 2) * 0.2 : 0;
   249→   249→  const noise = (Math.random() - 0.5) * config.variance;
   250→   250→  let value = config.base * (1 + seasonalFactor) + noise;
   251→   251→  // Clamp to sensor range
   252→   252→  value = Math.max(sensor.minRange, Math.min(sensor.maxRange, value));
   253→   253→  return Math.round(value * 100) / 100;
   254→   254→}

   401→   401→    // Certify batch
   402→   402→    batchJob.status = 'CERTIFYING';
   403→   403→    batchJob.certificationStatus = batchJob.rejectedRecords === 0 ? 'CERTIFIED' : (batchJob.validRecords > batchJob.rejectedRecords ? 'PARTIAL' : 'REJECTED');
   404→   404→    batchJob.merkleRoot = generateHash(batchJob.id, batchJob.validRecords, batchJob.createdAt);
   405→   405→    batchJob.status = 'COMPLETED';
   406→   406→    batchJob.completedAt = new Date().toISOString();
   407→   407→
   408→   408→    io.emit('iot:batch:completed', batchJob);
   409→   409→    io.emit('iot:readings', batchReadings);
   410→   410→  });

   455→   455→// ─── Simulation Engine ────────────────────────────────────────────────
   456→   456→function startSimulation() {
   457→   457→  if (simInterval) clearInterval(simInterval);
   458→   458→  simInterval = setInterval(() => {
   459→   459→    if (!simRunning) return;
   460→   460→    tickCounter++;
   461→   461→
   462→   462→    const newReadings: SensorReading[] = [];
   463→   463→
   464→   464→    for (const edgeNode of EDGE_NODES) {
   465→   465→      if (edgeNode.status !== 'online') continue;
   466→   466→
   467→   467→      for (const sensor of edgeNode.sensors) {
   468→   468→        if (sensor.status !== 'active') continue;
   469→   469→
   470→   470→        // สุ่มค่าตาม sampling rate (เพื่อไม่ให้ทุก sensor ส่งทุก tick)
   471→   471→        if (tickCounter % Math.max(1, Math.floor(sensor.samplingRate / 10)) !== 0) continue;
   472→   472→
   473→   473→        const value = generateSensorValue(sensor.sensorType, sensor, tickCounter);
   474→   474→        const timestamp = new Date().toISOString();
   475→   475→        const qualityFlag = validateReading(sensor, value);
   476→   476→        const integrityHash = generateHash(sensor.sensorId, value, timestamp);
   477→   477→        const signature = generateSignature(edgeNode.nodeId, integrityHash);
   478→   478→
   479→   479→        const reading: SensorReading = {
   480→   480→          id: `RD-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   481→   481→          edgeNodeId: edgeNode.nodeId,
   482→   482→          sensorId: sensor.sensorId,
   483→   483→          sensorType: sensor.sensorType,
   484→   484→          value,
   485→   485→          unit: sensor.unit,
   486→   486→          timestamp,
   487→   487→          receivedAt: new Date().toISOString(),
   488→   488→          ingestionMode: 'realtime',
   489→   489→          qualityFlag,
   490→   490→          integrityHash,
   491→   491→          signature,
   492→   492→        };
   493→   493→
   494→   494→        newReadings.push(reading);
   495→   495→        edgeNode.totalReadings++;
   496→   496→      }
   497→   497→
   498→   498→      // Update heartbeat
   499→   499→      edgeNode.lastHeartbeat = new Date().toISOString();
   500→   500→    }
   501→   501→
   502→   502→    if (newReadings.length > 0) {
   503→   503→      readings.push(...newReadings);
   504→   504→      // เก็บแค่ 2000 รายการล่าสุด
   505→   505→      if (readings.length > 2000) readings = readings.slice(-2000);
   506→   506→
   507→   507→      // Broadcast
   508→   508→      io.emit('iot:readings', newReadings);
   509→   509→    }
   510→   510→
   511→   511→    // ส่ง status update
   512→   512→    io.emit('iot:status', {
   513→   513→      running: simRunning,
   514→   514→      tick: tickCounter,
   515→   515→      edgeNodes: EDGE_NODES,
   516→   516→      totalReadings: readings.length,
   517→   517→      batchJobs: batchJobs.length,
   518→   518→      connectedClients: realtimeClients.size,
   519→   519→    });
   520→   520→
   521→   521→  }, 2000); // ทุก 2 วินาที
   522→   522→}
