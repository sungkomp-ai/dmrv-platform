    95→    95→export default function LidarFieldMap({
    96→    96→  fields, latestReadings, selectedFieldId, onFieldSelect, standardMode,
    97→    97→}: LidarFieldMapProps) {
    98→    98→  const mounted = useMounted();

   100→   100→  const mapRef = useRef<L.Map | null>(null);
   101→   101→  const tileLayerRef = useRef<L.TileLayer | null>(null);
   102→   102→  const labelLayerRef = useRef<L.TileLayer | null>(null);
   103→   103→  const fieldGroupRef = useRef<L.LayerGroup | null>(null);
   104→   104→  const markerGroupRef = useRef<L.LayerGroup | null>(null);

   106→   106→  const prevFieldsJson = useRef<string>('');
   107→   107→  const [isSatellite, setIsSatellite] = useState(true);
   108→   108→

   110→   110→  useEffect(() => {
   111→   111→    const styleId = 'lidar-map-animations';
   112→   112→    if (document.getElementById(styleId)) return;
   113→   113→    const style = document.createElement('style');
   114→   114→    style.id = styleId;
   115→   115→    style.textContent = `
   116→   116→      @keyframes lidar-pulse {
   117→   117→        0% { transform: scale(1); opacity: 0.7; }
   118→   118→        50% { transform: scale(1.5); opacity: 0.2; }
   119→   119→        100% { transform: scale(1); opacity: 0.7; }
   120→   120→      }
   121→   121→      @keyframes lidar-scan {
   122→   122→        0% { stroke-dashoffset: 0; }
   123→   123→        100% { stroke-dashoffset: 40; }
   124→   124→      }

   126→   126→        0% { transform: scale(0.8); opacity: 0.9; }
   127→   127→        50% { transform: scale(1.4); opacity: 0.3; }
   128→   128→        100% { transform: scale(0.8); opacity: 0.9; }
   129→   129→      }
   130→   130→      .lidar-alert-marker {
   131→   131→        animation: alert-pulse 1.5s ease-in-out infinite;
   132→   132→      }
   133→   133→    `;
   134→   134→    document.head.appendChild(style);
   135→   135→    return () => { style.remove(); };
   136→   136→  }, []);
   137→   137→
   138→   138→  // ─── Initialize map ────────────────────────────────────────────────
   139→   139→  useEffect(() => {
   140→   140→    if (!containerRef.current || mapRef.current) return;
   141→   141→
   142→   142→    const map = L.map(containerRef.current, {

   144→   144→      zoom: 6,
   145→   145→      zoomControl: false,
   146→   146→    });
   147→   147→
   148→   148→    L.control.zoom({ position: 'topright' }).addTo(map);
   149→   149→
   150→   150→    tileLayerRef.current = L.tileLayer(SATELLITE_URL, {
   151→   151→      attribution: '© Esri',
   152→   152→      maxZoom: 19,
   153→   153→    }).addTo(map);
   154→   154→
   155→   155→    labelLayerRef.current = L.tileLayer(SATELLITE_LABELS_URL, {
   156→   156→      maxZoom: 19,
   157→   157→    }).addTo(map);
   158→   158→
   159→   159→    fieldGroupRef.current = L.layerGroup().addTo(map);
   160→   160→    markerGroupRef.current = L.layerGroup().addTo(map);
   161→   161→    alertGroupRef.current = L.layerGroup().addTo(map);
   162→   162→
   163→   163→    mapRef.current = map;
   164→   164→
   165→   165→    return () => {
   166→   166→      map.remove();
   167→   167→      mapRef.current = null;
   168→   168→    };
   169→   169→  }, []);
   170→   170→
   171→   171→  // ─── Toggle satellite/street ───────────────────────────────────────
   172→   172→  useEffect(() => {
   173→   173→    if (!mapRef.current || !tileLayerRef.current) return;
   174→   174→
   175→   175→    mapRef.current.removeLayer(tileLayerRef.current);
   176→   176→    if (labelLayerRef.current) mapRef.current.removeLayer(labelLayerRef.current);
   177→   177→
   178→   178→    if (isSatellite) {
   179→   179→      tileLayerRef.current = L.tileLayer(SATELLITE_URL, {
   180→   180→        attribution: '© Esri',
   181→   181→        maxZoom: 19,
   182→   182→      }).addTo(mapRef.current);
   183→   183→      labelLayerRef.current = L.tileLayer(SATELLITE_LABELS_URL, {
   184→   184→        maxZoom: 19,
   185→   185→      }).addTo(mapRef.current);
   186→   186→    } else {
   187→   187→      tileLayerRef.current = L.tileLayer(STREET_URL, {
   188→   188→        attribution: '© OpenStreetMap',
   189→   189→        maxZoom: 19,
   190→   190→      }).addTo(mapRef.current);
   191→   191→      labelLayerRef.current = null;

   193→   193→  }, [isSatellite]);

   195→   195→  // ─── Update fields on map ──────────────────────────────────────────

   197→   197→    if (!fieldGroupRef.current || !markerGroupRef.current || !alertGroupRef.current) return;
   198→   198→
   199→   199→    const fieldsJson = JSON.stringify(fields.map(f => ({
   200→   200→      id: f.id, boundary: f.boundary, surveyPhase: f.surveyPhase,
   201→   201→      chmMeanHeight: f.chmMeanHeight, canopyCoverPct: f.canopyCoverPct,
   202→   202→      agbTonnePerHa: f.agbTonnePerHa, carbonTcPerHa: f.carbonTcPerHa,
   203→   203→      totalCO2eStock: f.totalCO2eStock, deforestationHa: f.deforestationHa,
   204→   204→      annualCarbonChangeTcPerHa: f.annualCarbonChangeTcPerHa,

   227→   227→          (c: number[]) => [c[0], c[1]] as L.LatLngExpression

   230→   230→        // Fill style depends on survey phase

   238→   238→          fillOpacity = 0.15;
   239→   239→          dashArray = '12, 6';
   240→   240→          fillColor = cfg.fill;
   241→   241→          weight = isSelected ? 4 : 2.5;
   242→   242→          borderColor = isSelected ? phaseColor : cfg.stroke;
   243→   243→        } else if (field.surveyPhase === 'PROCESSING') {
   244→   244→          fillOpacity = 0.25;
   245→   245→          dashArray = '8, 4';
   246→   246→          fillColor = cfg.fill;
   247→   247→          weight = isSelected ? 3.5 : 2;
   248→   248→          borderColor = isSelected ? phaseColor : cfg.stroke;
   249→   249→        } else {
   250→   250→          // VERIFIED

   259→   259→          color: borderColor,

   262→   262→          fillColor,

   265→   265→        });
   266→   266→

   268→   268→        const progress = Math.min((field.simMonth / 12) * 100, 100);

   270→   270→        const isTgo = standardMode === 'tgo';
   271→   271→        const carbonLabel = isTgo ? 'T-VER Carbon' : 'IPCC Carbon';
   272→   272→        const carbonValue = isTgo
   273→   273→          ? (reading?.tverCarbonTcPerHa?.toFixed(2) ?? field.carbonTcPerHa.toFixed(2))

   275→   275→        const co2eValue = isTgo
   276→   276→          ? (reading?.tverCo2eTonnePerHa?.toFixed(1) ?? field.totalCO2eStock.toFixed(0))

   278→   278→        const methodology = isTgo ? 'T-VER (Thailand Verified Emission Reduction)' : 'IPCC 2006 GL Vol 4';

   280→   280→        const carbonChangeSign = carbonChange >= 0 ? '+' : '';

   282→   282→

   284→   284→          <div style="min-width: 280px; font-family: system-ui, -apple-system, sans-serif;">

   291→   291→              ">${cfg.icon}</div>
   292→   292→              <div style="min-width: 0;">

   303→   303→              ${field.isMangrove ? '<span style="background: #ecfeff; color: #0e7490; padding: 3px 10px; border-radius: 99px; font-size: 10px; font-weight: 600; border: 1px solid #a5f3fc;">🌊 Mangrove</span>' : ''}
   304→   304→              ${hasDeforestation ? '<span style="background: #fef2f2; color: #dc2626; padding: 3px 10px; border-radius: 99px; font-size: 10px; font-weight: 600; border: 1px solid #fecaca;">⚠️ Deforestation</span>' : ''}
   305→   305→            </div>
   306→   306→
   307→   307→            <!-- Year/Month progress bar -->

   309→   309→              <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 4px;">

   312→   312→              </div>
   313→   313→              <div style="background: #e5e7eb; border-radius: 99px; height: 6px; overflow: hidden;">

   317→   317→

't/ไร่' : 't/ha'}</span></div>

   330→   330→                <div style="font-weight: 800; color: #1d4ed8; font-size: 16px; margin-top: 2px;">${standardMode === 'tgo' ?    349→   349→
   350→   350→            ${hasDeforestation ? `
