   433→function pick<T>(arr: T[]): T {
   434→  return arr[Math.floor(Math.random() * arr.length)];
   435→}

   437→let seedingPromise: Promise<void> | null = null;

   439→export async function ensureSeeded() {
   440→  if (seedingPromise) {
   441→    await seedingPromise;
   442→    return;
   443→  }
   444→
   445→  const projectCount = await db.project.count();
   446→  if (projectCount > 0) return;
   447→
   448→  seedingPromise = seedAll().finally(() => {
   449→    seedingPromise = null;
   450→  });
   451→  await seedingPromise;
   452→}
