    58→    58→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    59→    59→  // Thai chars are roughly same width as CJK
    60→    60→  const charWidth = (pt) => pt * 20;
    61→    61→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    62→    62→  let titlePt = preferredPt;
    63→    63→  let lines;
    64→    64→  while (titlePt >= minPt) {
    65→    65→    const cpl = charsPerLine(titlePt);
    66→    66→    if (cpl < 2) { titlePt -= 2; continue; }
    67→    67→    lines = splitTitleLines(title, cpl);
    68→    68→    if (lines.length <= 3) break;
    69→    69→    titlePt -= 2;
    70→    70→  }
    71→    71→  if (!lines || lines.length > 3) {
    72→    72→    const cpl = charsPerLine(minPt);
    73→    73→    lines = splitTitleLines(title, cpl);
    74→    74→    titlePt = minPt;
    75→    75→  }
    76→    76→  return { titlePt, titleLines: lines };
    77→    77→}
    78→    78→
    79→    79→function splitTitleLines(title, charsPerLine) {
    80→    80→  if (title.length <= charsPerLine) return [title];
    81→    81→  const breakAfter = new Set([
    82→    82→    ..." \t", "-", "/",
    83→    83→  ]);
    84→    84→  const lines = [];
    85→    85→  let remaining = title;
    86→    86→  while (remaining.length > charsPerLine) {
    87→    87→    let breakAt = -1;
    88→    88→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    89→    89→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) {
    90→    90→        breakAt = i; break;
    91→    91→      }
    92→    92→    }
    93→    93→    if (breakAt === -1) {
    94→    94→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    95→    95→      for (let i = charsPerLine + 1; i < limit; i++) {
    96→    96→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    97→    97→      }
    98→    98→    }
    99→    99→    if (breakAt === -1) breakAt = charsPerLine;
   100→   100→    lines.push(remaining.slice(0, breakAt).trim());
   101→   101→    remaining = remaining.slice(breakAt).trim();
   102→   102→  }
   103→   103→  if (remaining) lines.push(remaining);
   104→   104→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
   105→   105→    const last = lines.pop();
   106→   106→    lines[lines.length - 1] += last;
   107→   107→  }
   108→   108→  return lines;
   109→   109→}

   111→   111→function calcCoverSpacing(params) {
   112→   112→  const {
   113→   113→    titleLineCount = 1, titlePt = 36, hasSubtitle = false,
   114→   114→    hasEnglishLabel = false, metaLineCount = 0,
   115→   115→    fixedHeight = 800, pageHeight = 16838,
   116→   116→    marginTop = 0, marginBottom = 0,
   117→   117→  } = params;
   118→   118→  const SAFETY = 1200;
   119→   119→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
   120→   120→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
   121→   121→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
   122→   122→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
   123→   123→  const metaHeight = metaLineCount * (10 * 23 + 100);
   124→   124→  const implicitParaHeight = 3 * 300;
   125→   125→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight +
   126→   126→    metaHeight + fixedHeight + implicitParaHeight;
   127→   127→  const remainingSpace = usableHeight - contentHeight;
   128→   128→  const safeRemaining = Math.max(remainingSpace, 400);
   129→   129→  const FOOTER_MIN = 800;
   130→   130→  const rawTop = Math.floor(safeRemaining * 0.45);
   131→   131→  const rawBottom = Math.floor(safeRemaining * 0.45);
   132→   132→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
   133→   133→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   134→   134→  return { topSpacing, midSpacing: Math.max(safeRemaining - topSpacing - bottomSpacing, 0), bottomSpacing };
   135→   135→}
