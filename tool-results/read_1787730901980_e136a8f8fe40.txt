   103→   103→function fmtTime(dateStr: string | null): string {
   104→   104→  if (!dateStr) return 'Never';
   105→   105→  const d = new Date(dateStr), now = new Date();
   106→   106→  const mins = Math.floor((now.getTime() - d.getTime()) / 60000);
   107→   107→  if (mins < 1) return 'Just now';
   108→   108→  if (mins < 60) return `${mins}m ago`;
   109→   109→  const hrs = Math.floor(mins / 60);
   110→   110→  if (hrs < 24) return `${hrs}h ago`;
   111→   111→  const days = Math.floor(hrs / 24);
   112→   112→  return days < 30 ? `${days}d ago` : d.toLocaleDateString();
   113→   113→}
