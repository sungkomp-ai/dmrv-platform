   135→// ─── Component ──────────────────────────────────────────────────────
   136→export default function LidarSimulatorView() {
   137→  const mounted = useMounted();
   138→  const [socket, setSocket] = useState<Socket | null>(null);
   139→  const [connected, setConnected] = useState(false);
   140→  const [status, setStatus] = useState<LidarSimStatus | null>(null);
   141→  const [readings, setReadings] = useState<LidarReading[]>([]);
   142→  const [sendingToDmrv, setSendingToDmrv] = useState(false);
   143→  const [dMRVCount, setDMRVCount] = useState(0);
   144→  const [selectedFieldId, setSelectedFieldId] = useState<string | null>(null);
   145→  const [standardMode, setStandardMode] = useState<'international' | 'tgo'>('tgo');
   146→
   147→  // Connect to simulator
   148→  useEffect(() => {
   149→    if (!mounted) return;
   150→    let s: Socket | null = null;
   151→    try {
   152→      s = io('/?XTransformPort=3033', { transports: ['polling', 'websocket'], reconnectionAttempts: 5, reconnectionDelay: 3000, timeout: 8000 });
   153→      s.on('connect', () => setConnected(true));
   154→      s.on('disconnect', () => setConnected(false));
   155→      s.on('lidar:status', (data: LidarSimStatus) => setStatus(data));
   156→      s.on('lidar:reading', (data: LidarReading[]) => {
   157→        setReadings(prev => [...prev, ...data].slice(-500));
   158→      });
   159→      s.on('lidar:readings', (data: LidarReading[]) => {
   160→        setReadings(data.slice(-500));
   161→      });
   162→      s.on('lidar:started', () => {});
   163→      s.on('lidar:stopped', () => {});
   164→      s.on('lidar:reset', () => setReadings([]));
   165→      setSocket(s);
   166→    } catch {
   167→      // Fallback: poll via HTTP
   168→    }
   169→
   170→    // HTTP polling fallback — fetch status every 3s if WebSocket fails
   171→    const pollInterval = setInterval(async () => {
   172→      try {
   173→        const res = await fetch('/api/status?XTransformPort=3033');
   174→        if (res.ok) {
   175→          const data = await res.json();
   176→          setStatus(data);
   177→          setConnected(true);
   178→        }
   179→        const readingsRes = await fetch('/api/readings?limit=50&XTransformPort=3033');
   180→        if (readingsRes.ok) {
   181→          const rData = await readingsRes.json();
   182→          const readingsArray = Array.isArray(rData) ? rData : (rData.readings || []);
   183→          if (readingsArray.length > 0) {
   184→            setReadings(prev => {
   185→              const lastTick = prev.length > 0 ? prev[prev.length - 1].tick : 0;
   186→              const newReadings = readingsArray.filter((r: LidarReading) => r.tick > lastTick);
   187→              if (newReadings.length > 0) return [...prev, ...newReadings].slice(-500);
   188→              return prev;
   189→            });
   190→          }
   191→        }
   192→      } catch { /* ignore */ }
   193→    }, 3000);
   194→
   195→    // Initial fetch
   196→    (async () => {
   197→      try {
   198→        const res = await fetch('/api/status?XTransformPort=3033');
   199→        if (res.ok) { const data = await res.json(); setStatus(data); setConnected(true); }
   200→      } catch { /* ignore */ }
   201→    })();
   202→
   203→    return () => {
   204→      if (s) s.disconnect();
   205→      clearInterval(pollInterval);
   206→    };
   207→  }, [mounted]);
   208→
   209→  const startSim = useCallback(async () => {
   210→    if (socket?.connected) { socket.emit('lidar:start'); return; }
   211→    try {
   212→      await fetch('/api/control?XTransformPort=3033', {
   213→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ action: 'start' }),
   215→      });
   216→    } catch { /* ignore */ }
   217→  }, [socket]);
   218→  const stopSim = useCallback(async () => {
   219→    if (socket?.connected) { socket.emit('lidar:stop'); return; }
   220→    try {
   221→      await fetch('/api/control?XTransformPort=3033', {
   222→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({ action: 'stop' }),
   224→      });
   225→    } catch { /* ignore */ }
   226→  }, [socket]);
   227→  const resetSim = useCallback(async () => {
   228→    if (socket?.connected) { socket.emit('lidar:reset'); setReadings([]); return; }
   229→    try {
   230→      await fetch('/api/control?XTransformPort=3033', {
   231→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ action: 'reset' }),
   233→      });
   234→    } catch { /* ignore */ }
   235→    setReadings([]);
   236→  }, [socket]);
   237→
   238→  // Auto-start when connected
   239→  useEffect(() => {
   240→    if (connected && status && !status.running) {
   241→      const timer = setTimeout(() => startSim(), 1000);
   242→      return () => clearTimeout(timer);
   243→    }
   244→  }, [connected, status?.running, startSim]);
   245→
   246→  // Send to dMRV
   247→  const sendToDmrv = useCallback(async () => {
   248→    if (readings.length === 0) return;
   249→    setSendingToDmrv(true);
   250→    try {
   251→      const res = await fetch('/api/dmrv/lidar', {
   252→        method: 'POST',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify({ action: 'ingest_batch', readings: readings.slice(-50) }),
   255→      });
   256→      if (res.ok) {
   257→        const data = await res.json();
   258→        setDMRVCount(prev => prev + (data.ingested || 0));
   259→      }
   260→    } finally { setSendingToDmrv(false); }
   261→  }, [readings]);
   262→
   263→  // ─── Computed Data ────────────────────────────────────────────────
   264→  const fieldIds = useMemo(() => ['khao-yai', 'mae-wong', 'samut-mangrove'], []);
   265→
   266→  const fieldReadings = useMemo(() => {
   267→    const map: Record<string, LidarReading[]> = {};
   268→    for (const fid of fieldIds) {
   269→      map[fid] = readings.filter(r => r.fieldId === fid).slice(-200);
   270→    }
   271→    return map;
   272→  }, [readings, fieldIds]);
   273→
   274→  const latestByField = useMemo(() => {
   275→    const map: Record<string, LidarReading | null> = {};
   276→    for (const fid of fieldIds) {
   277→      const arr = fieldReadings[fid];
   278→      map[fid] = arr.length > 0 ? arr[arr.length - 1] : null;
   279→    }
   280→    return map;
   281→  }, [fieldReadings, fieldIds]);
   282→
   283→  // Chart data - grouped by tick
   284→  const chartData = useMemo(() => {
   285→    const tickMap: Record<number, Record<string, LidarReading>> = {};
   286→    for (const r of readings) {
   287→      if (!tickMap[r.tick]) tickMap[r.tick] = {};
   288→      tickMap[r.tick][r.fieldId] = r;
   289→    }
   290→    return Object.entries(tickMap)
   291→      .sort(([a], [b]) => Number(a) - Number(b))
   292→      .slice(-100)
   293→      .map(([tick, fields]) => {
   294→        const entry: Record<string, number | string> = { tick: Number(tick), month: 0, year: 0 };
   295→        for (const [fid, r] of Object.entries(fields)) {
   296→          const cfg = FOREST_CONFIG[fid];
   297→          if (!cfg) continue;
   298→          const key = cfg.shortLabel;
   299→          // Carbon stock (convert to per-rai in TGO mode)
   300→          entry[`${key}_carbon`] = standardMode === 'tgo' ? (r.tverCarbonTcPerHa ?? r.carbonTcPerHa) * 0.16 : (r.tverCarbonTcPerHa ?? r.carbonTcPerHa);
   301→          entry[`${key}_co2e`] = standardMode === 'tgo' ? (r.tverCo2eTonnePerHa ?? r.co2eTonnePerHa) : r.co2eTonnePerHa;
   302→          entry[`${key}_totalCarbon`] = r.totalCarbonStockTc;
   303→          entry[`${key}_totalCO2e`] = r.totalCO2eStock;
   304→          // Biomass (convert to per-rai in TGO mode)
   305→          entry[`${key}_agb`] = standardMode === 'tgo' ? r.agbTonnePerHa * 0.16 : r.agbTonnePerHa;
   306→          entry[`${key}_bgb`] = standardMode === 'tgo' ? r.bgbTonnePerHa * 0.16 : r.bgbTonnePerHa;
   307→          // Change detection (convert to per-rai in TGO mode)
   308→          entry[`${key}_change`] = standardMode === 'tgo' ? r.annualCarbonChangeTcPerHa * 0.16 : r.annualCarbonChangeTcPerHa;
   309→          entry[`${key}_deforest`] = standardMode === 'tgo' ? r.deforestationHa * 6.25 : r.deforestationHa;
   310→          // LIDAR quality
   311→          entry[`${key}_pointDensity`] = r.pointDensityPerM2;
   312→          entry[`${key}_scanCoverage`] = r.scanCoveragePct;
   313→          // Structural metrics
   314→          entry[`${key}_height`] = r.chmMeanHeight;
   315→          entry[`${key}_canopy`] = r.canopyCoverPct;
   316→          entry[`${key}_lai`] = r.lai;
   317→          // Soil carbon (convert to per-rai in TGO mode)
   318→          entry[`${key}_soilCarbon`] = standardMode === 'tgo' ? r.soilCarbonTcPerHa * 0.16 : r.soilCarbonTcPerHa;
   319→          // TGO uncertainty
   320→          entry[`${key}_uncertainty`] = r.tverUncertaintyPct ?? 0;
   321→          entry['month'] = r.simMonth;
   322→          entry['year'] = r.simYear;
   323→        }
   324→        return entry;
   325→      });
   326→  }, [readings, standardMode]);
   327→
   328→  // Summary stats
   329→  const totalReadings = readings.length;
   330→  const currentYear = status?.simYear ?? 0;
   331→  const currentMonth = status?.simMonth ?? 0;
   332→  const totalCarbonStock = useMemo(() => {
   333→    if (!status?.fields) return 0;
   334→    return status.fields.reduce((s, f) => s + (f.totalCarbonStockTc || 0), 0);
   335→  }, [status?.fields]);
   336→  const totalCO2eStock = useMemo(() => {
   337→    if (!status?.fields) return 0;
   338→    return status.fields.reduce((s, f) => s + (f.totalCO2eStock || 0), 0);
   339→  }, [status?.fields]);
   340→  const deforestationAlerts = useMemo(() => {
   341→    if (!status?.fields) return 0;
   342→    return status.fields.filter(f => f.deforestationHa > 0).length;
   343→  }, [status?.fields]);
   344→
   345→  // Year progress (0-5 years)
   346→  const yearProgress = useMemo(() => {
   347→    if (!status) return 0;
   348→    const totalMonths = (status.totalYears || 5) * 12;
   349→    const elapsedMonths = (status.simYear || 0) * 12 + (status.simMonth || 0);
   350→    return Math.min((elapsedMonths / totalMonths) * 100, 100);
   351→  }, [status]);
   352→
   353→  if (!mounted) {
   354→    return <div className="flex items-center justify-center h-64"><RefreshCw className="h-8 w-8 animate-spin text-emerald-600" /></div>;
   355→  }
   356→
   357→  return (
   358→    <div className="space-y-6">
   359→      {/* Hero Header */}
   360→      <div className="relative overflow-hidden rounded-xl bg-gradient-to-r from-emerald-800 via-teal-700 to-cyan-700 p-6 text-white">
   361→        <div className="absolute inset-0 opacity-10">
   362→          <div className="absolute top-4 right-10 text-8xl">🌴</div>
   363→          <div className="absolute bottom-2 left-20 text-6xl">🌲</div>
   364→        </div>
   365→        <div className="relative z-10">
   366→          <div className="flex items-center gap-3 mb-2 flex-wrap">
   367→            <Trees className="h-7 w-7" />
   368→            <h2 className="text-2xl font-bold">LIDAR Carbon Credit Verification</h2>
   369→            <Badge className="bg-white/20 text-white hover:bg-white/30 text-xs">5-Year Monitoring</Badge>
   370→            <Badge className="bg-amber-500/80 text-white text-[9px]">T-VER-P-FOR-001</Badge>
   371→            <div className="flex items-center gap-1 bg-white/10 rounded-lg p-0.5 text-[10px]">
   372→              <button
   373→                onClick={() => setStandardMode('tgo')}
   374→                className={`px-2.5 py-1 rounded-md transition-all ${standardMode === 'tgo' ? 'bg-white/25 font-bold' : 'hover:bg-white/10'}`}
   375→              >
   376→                🇹🇭 TGO/T-VER
   377→              </button>
   378→              <button
   379→                onClick={() => setStandardMode('international')}
   380→                className={`px-2.5 py-1 rounded-md transition-all ${standardMode === 'international' ? 'bg-white/25 font-bold' : 'hover:bg-white/10'}`}
   381→              >
   382→                🌐 International
   383→              </button>
   384→            </div>
   385→          </div>
   386→          <p className="text-sm text-white/80 mb-1">
   387→            LIDAR-based forest carbon stock monitoring across 3 Thai forest areas with 5-year verification cycle
   388→          </p>
   389→          <p className="text-[10px] text-white/60 mb-4">
   390→            {standardMode === 'tgo'
   391→              ? 'อ้างอิงมาตรฐาน: T-VER-P-FOR-001 (ป่าไม้) / T-VER-BC-001 (Blue Carbon) | Thai allometric equations (Ogawa et al.) | CF: 0.47 evergreen, 0.49 mangrove | หน่วย: tC/ไร่, tCO₂e, พื้นที่: ไร่ (1 ไร่ = 0.16 ha)'
   392→              : 'Reference: IPCC 2006 GL Vol. 4 | Verra VCS VM0047 | Gold Standard | IPCC allometric equations | CF = 0.47 default | ISO 14064-2 | Units: tC/ha, tCO₂e/ha'
   393→            }
   394→          </p>
   395→          <div className="flex flex-wrap items-center gap-3">
   396→            <Button size="sm" onClick={startSim} disabled={status?.running} className="bg-white/20 hover:bg-white/30 text-white border-white/30">
   397→              <Play className="h-3.5 w-3.5 mr-1" /> Start
   398→            </Button>
   399→            <Button size="sm" variant="outline" onClick={stopSim} disabled={!status?.running} className="border-white/30 text-white hover:bg-white/20">
   400→              <Square className="h-3.5 w-3.5 mr-1" /> Stop
   401→            </Button>
   402→            <Button size="sm" variant="outline" onClick={resetSim} className="border-white/30 text-white hover:bg-white/20">
   403→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> Reset
   404→            </Button>
   405→            <div className="w-px h-6 bg-white/30" />
   406→            <Button size="sm" onClick={sendToDmrv} disabled={sendingToDmrv || readings.length === 0}
   407→              className="bg-emerald-500 hover:bg-emerald-400 text-white">
   408→              <Send className="h-3.5 w-3.5 mr-1" /> Send to dMRV
   409→            </Button>
   410→            <div className="flex items-center gap-2 ml-auto text-xs text-white/70">
   411→              <div className={`h-2 w-2 rounded-full ${connected ? 'bg-green-400 animate-pulse' : 'bg-red-400'}`} />
   412→              {connected ? 'Connected' : 'Disconnected'}
   413→              {status?.running && <Badge className="bg-green-500/50 text-white text-[9px] animate-pulse">LIVE</Badge>}
   414→            </div>
   415→          </div>
   416→          {/* Year/Month Progress */}
   417→          <div className="mt-3">
   418→            <div className="flex items-center justify-between text-xs text-white/70 mb-1">
   419→              <span>Year {currentYear} / {status?.totalYears ?? 5} &middot; Month {currentMonth}/12</span>
   420→              <span>1 sec = 1 month simulated</span>
   421→            </div>
   422→            <Progress value={yearProgress} className="h-1.5 bg-white/20" />
   423→          </div>
   424→        </div>
   425→      </div>
   426→
   427→      {/* KPI Cards */}
   428→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   429→        {[
   430→          { icon: Satellite, label: 'LIDAR Scans', value: totalReadings.toLocaleString(), color: 'text-cyan-600', bg: 'bg-cyan-50' },
   431→          { icon: Clock, label: 'Survey Year', value: `${currentYear}/${status?.totalYears ?? 5}`, color: 'text-emerald-600', bg: 'bg-emerald-50' },
   432→          { icon: Trees, label: standardMode === 'tgo' ? 'Total Carbon (tC)' : 'Total Carbon (tC)', value: totalCarbonStock.toFixed(1), color: 'text-green-600', bg: 'bg-green-50' },
   433→          { icon: Leaf, label: 'Total CO\u2082e', value: totalCO2eStock.toFixed(1), color: 'text-teal-600', bg: 'bg-teal-50' },
   434→          { icon: AlertTriangle, label: 'Deforestation', value: deforestationAlerts, color: 'text-red-600', bg: 'bg-red-50' },
   435→          { icon: Send, label: 'dMRV Sent', value: dMRVCount, color: 'text-purple-600', bg: 'bg-purple-50' },
   436→        ].map(s => (
   437→          <Card key={s.label} className="shadow-none">
   438→            <CardContent className="p-3 flex items-center gap-3">
   439→              <div className={`h-9 w-9 rounded-xl ${s.bg} flex items-center justify-center shrink-0`}>
   440→                <s.icon className={`h-4 w-4 ${s.color}`} />
   441→              </div>
   442→              <div>
   443→                <p className="text-[10px] text-muted-foreground font-medium">{s.label}</p>
   444→                <p className="text-lg font-bold leading-none">{s.value}</p>
   445→              </div>
   446→            </CardContent>
   447→          </Card>
   448→        ))}
   449→      </div>
