   269→   269→function generateSampleMeasurements(
   270→   270→  plotId: string,
   271→   271→  plotData: typeof SAMPLE_PLOTS[0],
   272→   272→  terraceIds?: string[]
   273→   273→) {
   274→   274→  const baseDate = new Date('2024-01-15');
   275→   275→  const measurements = [];
   276→   276→
   277→   277→  // สร้างข้อมูลการวัด 3 ครั้งต่อแปลง (ทุก 4 เดือน)
   278→   278→  for (let i = 0; i < 3; i++) {
   279→   279→    const date = new Date(baseDate.getTime() + i * 120 * 24 * 60 * 60 * 1000);
   280→   280→    const isAnomaly = i === 2 && plotId === 'SOIL-KKN-002'; // แปลงขอนแก่น มีข้อมูลผิดปกติในรอบสุดท้าย
   281→   281→
   282→   282→    // ข้อมูลตะกอนดินตามความลาดชัน
   283→   283→    const slopeFactor = plotData.slopeDegree / 20;
   284→   284→    const runoffVolumeL = Math.round((500 + slopeFactor * 2000 + Math.random() * 500) * 10) / 10;
   285→   285→    const sedimentConcGpL = Math.round((5 + slopeFactor * 15 + Math.random() * 5) * 100) / 100;
   286→   286→
   287→   287→    measurements.push({
   288→   288→      plotId,
   289→   289→      measurementDate: date.toISOString(),
   290→   290→      measurementType: i === 1 ? 'LAB' : 'FIELD',
   291→   291→      depthCm: 30,
   292→   292→      soilMoisture: isAnomaly ? 95 : 30 + Math.random() * 15,
   293→   293→      soilTemperature: 25 + Math.random() * 5,
   294→   294→      soilPH: 6.0 + Math.random() * 0.8,
   295→   295→      bulkDensity: isAnomaly ? 3.2 : 1.1 + Math.random() * 0.3,
   296→   296→      organicCarbonPct: plotData.baselineSOC / 30 + (i * 0.4) + Math.random() * 0.2,
   297→   297→      totalNitrogen: 0.12 + Math.random() * 0.05,
   298→   298→      cec: 12 + Math.random() * 8,
   299→   299→      aboveGroundBiomass: 3 + Math.random() * 2,
   300→   300→      belowGroundBiomass: 1.2 + Math.random() * 0.8,
   301→   301→      co2Flux: 2 + Math.random() * 3,
   302→   302→      ch4Flux: plotData.cropType === 'Rice' ? 5 + Math.random() * 10 : 0.5 + Math.random() * 2,
   303→   303→      n2oFlux: 10 + Math.random() * 30,
   304→   304→      runoffVolumeL,
   305→   305→      sedimentConcGpL,
   306→   306→      sedimentLoadKg: Math.round(runoffVolumeL * sedimentConcGpL / 1000 * 1000) / 1000,
   307→   307→      erosionRateTpHa: Math.round((plotData.baselineSediment / 3 + Math.random() * 3) * 1000) / 1000,
   308→   308→      waterFlowRateLMin: Math.round((5 + slopeFactor * 15 + Math.random() * 5) * 10) / 10,
   309→   309→      infiltrationMmHr: Math.round((20 + Math.random() * 30) * 10) / 10,
   310→   310→      ingestionMode: i === 2 ? 'batch' : 'realtime',
   311→   311→      qualityFlag: isAnomaly ? 'ANOMALY' : 'VALIDATED',
   312→   312→      sourceDevice: i === 1 ? `LAB-${plotId.split('-')[1]}-001` : `SENSOR-${plotId.split('-')[1]}-001`,
   313→   313→      notes: isAnomaly
   314→   314→        ? 'ค่า bulkDensity ผิดปกติ กรุณาตรวจสอบ'
   315→   315→        : `การวัดครั้งที่ ${i + 1} (พื้นที่ลาดชัน ${plotData.slopeDegree}°)`,
   316→   316→    });
   317→   317→

   321→   321→        measurements.push({
   322→   322→          plotId,
   323→   323→          terraceLayerId: terraceIds[t],
   324→   324→          measurementDate: date.toISOString(),
   325→   325→          measurementType: 'FIELD',
   326→   326→          depthCm: 30,
   327→   327→          soilMoisture: 32 + Math.random() * 12,
   328→   328→          soilTemperature: 24 + Math.random() * 5,
   329→   329→          soilPH: 6.2 + Math.random() * 0.6,
   330→   330→          bulkDensity: 1.1 + Math.random() * 0.2,
   331→   331→          organicCarbonPct: plotData.baselineSOC / 30 + (t * 0.2) + Math.random() * 0.1,

   333→   333→          cec: 14 + Math.random() * 6,
   334→   334→          aboveGroundBiomass: 2.5 + Math.random() * 1.5,
   335→   335→          belowGroundBiomass: 1.0 + Math.random() * 0.5,
   336→   336→          co2Flux: 2.5 + Math.random() * 2,
   337→   337→          ch4Flux: 1 + Math.random() * 3,
   338→   338→          n2oFlux: 12 + Math.random() * 20,
   339→   339→          runoffVolumeL: Math.round((200 + t * 100 + Math.random() * 100) * 10) / 10,
   340→   340→          sedimentConcGpL: Math.round((3 + plotData.slopeDegree / 10 + Math.random() * 3) * 100) / 100,
   341→   341→          sedimentLoadKg: 0, // จะคำนวณอัตโนมัติ

   343→   343→          waterFlowRateLMin: Math.round((3 + t * 2 + Math.random() * 3) * 10) / 10,

   345→   345→          ingestionMode: 'realtime',
   346→   346→          qualityFlag: 'VALIDATED',
