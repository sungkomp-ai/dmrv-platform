   303→function UserMenuButton({ onNavigateToUsers }: { onNavigateToUsers: () => void }) {
   304→  const router = useRouter();
   305→  const [user, setUser] = useState<CurrentUser | null>(null);
   306→  const [loading, setLoading] = useState(true);
   307→
   308→  const refresh = useCallback(async () => {
   309→    try {
   310→      const res = await fetch('/api/dmrv/me');
   311→      if (res.ok) {
   312→        const data = await res.json();
   313→        setUser(data.user || null);
   314→      } else {
   315→        setUser(null);
   316→      }
   317→    } catch {
   318→      setUser(null);
   319→    } finally {
   320→      setLoading(false);
   321→    }
   322→  }, []);
   323→
   324→  useEffect(() => {
   325→    refresh();
   326→    // Poll every 30s so logout from another tab reflects here
   327→    const t = setInterval(refresh, 30000);
   328→    return () => clearInterval(t);
   329→  }, [refresh]);
   330→
   331→  // Note: do NOT auto-redirect to /login when no user.
   332→  // The dashboard should render regardless — the UserMenuButton
   333→  // shows a "Sign in" button for unauthenticated users instead.
   334→
   335→  if (loading) {
   336→    return <div className="h-8 w-8 rounded-full bg-slate-200 animate-pulse" />;
   337→  }
   338→
   339→  if (!user) {
   340→    return (
   341→      <Button variant="outline" size="sm" className="gap-1.5" onClick={() => router.push('/login')}>
   342→        <LogIn className="h-3.5 w-3.5" />
   343→        <span className="hidden sm:inline">Sign in</span>
   344→      </Button>
   345→    );
   346→  }
   347→
   348→  const isDev = user.id.startsWith('dev-');
   349→  const initials = (user.name || user.username).charAt(0).toUpperCase();
   350→
   351→  return (
   352→    <Popover>
   353→      <PopoverTrigger asChild>
   354→        <Button variant="ghost" size="sm" className="gap-2 px-2">
   355→          <div className={`flex h-7 w-7 items-center justify-center rounded-full text-xs font-bold ${
   356→            user.role === 'admin' ? 'bg-red-100 text-red-700' :
   357→            user.role === 'validator' ? 'bg-violet-100 text-violet-700' :
   358→            user.role === 'buyer' ? 'bg-sky-100 text-sky-700' :
   359→            'bg-emerald-100 text-emerald-700'
   360→          }`}>
   361→            {initials}
   362→          </div>
   363→          <div className="hidden sm:flex flex-col items-start leading-tight">
   364→            <span className="text-xs font-semibold max-w-[100px] truncate">{user.name || user.username}</span>
   365→            <span className="text-[10px] text-muted-foreground">{user.role}{isDev ? ' · dev' : ''}</span>
   366→          </div>
   367→        </Button>
   368→      </PopoverTrigger>
   369→      <PopoverContent className="w-64 p-0" align="end">
   370→        <div className="border-b p-4">
   371→          <div className="flex items-center gap-3">
   372→            <div className={`flex h-10 w-10 items-center justify-center rounded-full text-sm font-bold ${
   373→              user.role === 'admin' ? 'bg-red-100 text-red-700' :
   374→              user.role === 'validator' ? 'bg-violet-100 text-violet-700' :
   375→              user.role === 'buyer' ? 'bg-sky-100 text-sky-700' :
   376→              'bg-emerald-100 text-emerald-700'
   377→            }`}>
   378→              {initials}
   379→            </div>
   380→            <div className="min-w-0">
   381→              <p className="text-sm font-semibold truncate">{user.name || user.username}</p>
   382→              <p className="text-[11px] text-muted-foreground truncate">{user.email}</p>
   383→            </div>
   384→          </div>
   385→          <div className="mt-3 flex flex-wrap gap-1.5">
   386→            <Badge variant="outline" className="text-[10px] capitalize">{user.role}</Badge>
   387→            {isDev ? (
   388→              <Badge variant="outline" className="text-[10px] bg-amber-50 text-amber-700 border-amber-200">dev fallback</Badge>
   389→            ) : (
   390→              <Badge variant="outline" className="text-[10px] bg-emerald-50 text-emerald-700 border-emerald-200">{user.status}</Badge>
   391→            )}
   392→          </div>
   393→        </div>
   394→        <div className="p-2">
   395→          <Button variant="ghost" size="sm" className="w-full justify-start gap-2 text-xs" onClick={onNavigateToUsers}>
   396→            <Users className="h-3.5 w-3.5" />
   397→            User Management
   398→          </Button>
   399→          <LogoutButton onLoggedOut={refresh} className="w-full justify-start gap-2 text-xs" />
   400→        </div>
   401→      </PopoverContent>
   402→    </Popover>
   403→  );
   404→}

   406→function SidebarContent({ collapsed = false, activeModule, onModuleChange, menuItemsWithBadges }: SidebarContentProps) {
   407→  const items = menuItemsWithBadges || menuItems;
   408→  const scrollRef = React.useRef<HTMLDivElement>(null);
   409→  const [canScrollDown, setCanScrollDown] = useState(false);
   410→  const [canScrollUp, setCanScrollUp] = useState(false);
   411→
   412→  // Pre-compute which items should show section headers
   413→  const itemsWithSections = items.map((item, idx) => {
   414→    const prevItem = idx > 0 ? items[idx - 1] : null;
   415→    const showSection = item.section && (!prevItem || prevItem.section !== item.section);
   416→    return { ...item, showSection };
   417→  });
   418→
   419→  const checkScroll = React.useCallback(() => {
   420→    const el = scrollRef.current;
   421→    if (el) {
   422→      setCanScrollUp(el.scrollTop > 4);
   423→      setCanScrollDown(el.scrollTop + el.clientHeight < el.scrollHeight - 4);
   424→    }
   425→  }, []);
   426→
   427→  useEffect(() => {
   428→    checkScroll();
   429→    window.addEventListener('resize', checkScroll);
   430→    return () => window.removeEventListener('resize', checkScroll);
   431→  }, [checkScroll, collapsed]);
   432→
   433→  // Auto-scroll to active item when it changes
   434→  useEffect(() => {
   435→    const timer = setTimeout(() => {
   436→      const el = scrollRef.current;
   437→      if (!el) return;
   438→      const activeBtn = el.querySelector('[data-active="true"]');
   439→      if (activeBtn) {
   440→        activeBtn.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
   441→      }
   442→      checkScroll();
   443→    }, 100);
   444→    return () => clearTimeout(timer);
   445→  }, [activeModule, checkScroll]);
   446→
   447→  return (
   448→    <div className="flex h-full flex-col bg-slate-900 text-white">
   449→      {/* Logo Area */}
   450→      <div className="flex items-center gap-3 px-4 py-5 border-b border-slate-700/50">
   451→        <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold text-sm">
   452→          dM
   453→        </div>
   454→        {!collapsed && (
   455→          <div className="flex flex-col overflow-hidden">
   456→            <span className="text-sm font-bold tracking-tight">dMRV Platform</span>
   457→            <span className="text-[11px] text-slate-400">Enterprise v4.0</span>
   458→          </div>
   459→        )}
   460→      </div>
   461→
   462→      {/* Scroll indicator top */}
   463→      {canScrollUp && (
   464→        <div className="relative h-0 z-10">
   465→          <div className="absolute inset-x-0 top-0 h-6 bg-gradient-to-b from-slate-900 to-transparent pointer-events-none" />
   466→        </div>
   467→      )}
   468→
   469→      {/* Navigation */}
   470→      <div
   471→        ref={scrollRef}
   472→        onScroll={checkScroll}
   473→        className="flex-1 px-2 py-3 overflow-y-auto sidebar-scroll"
   474→      >
   475→        <nav className="flex flex-col gap-0.5">
   476→          {itemsWithSections.map((item) => {
   477→            const isActive = activeModule === item.key;
   478→            const Icon = item.icon;
   479→
   480→            return (
   481→              <React.Fragment key={item.key}>
   482→                {item.showSection && !collapsed && (
   483→                  <div className="pt-3 pb-1 px-3">
   484→                    <span className="text-[9px] font-bold text-slate-500 uppercase tracking-widest">{item.section}</span>
   485→                  </div>
   486→                )}
   487→                {item.showSection && collapsed && (
   488→                  <div className="pt-2 pb-1 mx-2">
   489→                    <div className="border-t border-slate-700/50" />
   490→                  </div>
   491→                )}
   492→                <button
   493→                  data-active={isActive}
   494→                  onClick={() => onModuleChange(item.key)}
   495→                  className={`flex items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition-all duration-150 ${
   496→                    isActive
   497→                      ? 'bg-emerald-600/20 text-emerald-400 shadow-sm'
   498→                      : 'text-slate-300 hover:bg-slate-800 hover:text-white'
   499→                  } ${collapsed ? 'justify-center' : ''}`}
   500→                  title={collapsed ? item.label : undefined}
   501→                >
   502→                  <Icon className={`h-4 w-4 shrink-0 ${isActive ? 'text-emerald-400' : ''}`} />
   503→                  {!collapsed && (
   504→                    <>
   505→                      <span className="truncate">{item.label}</span>
   506→                      {item.badge && (
   507→                        <Badge className={`ml-auto h-5 min-w-[20px] justify-center text-[10px] text-white ${item.badgeColor || 'bg-emerald-600 hover:bg-emerald-600'}`}>
   508→                          {item.badge}
   509→                        </Badge>
   510→                      )}
   511→                    </>
   512→                  )}
   513→                  {isActive && !collapsed && !item.badge && (
   514→                    <div className="ml-auto h-1.5 w-1.5 rounded-full bg-emerald-400" />
   515→                  )}
   516→                </button>
   517→              </React.Fragment>
   518→            );
   519→          })}
   520→        </nav>
   521→      </div>
   522→
   523→      {/* Scroll indicator bottom */}
   524→      {canScrollDown && !collapsed && (
   525→        <div className="relative h-0 z-10 -mt-6">
   526→          <div className="absolute inset-x-0 bottom-0 h-8 bg-gradient-to-t from-slate-900 via-slate-900/90 to-transparent pointer-events-none flex items-end justify-center pb-1">
   527→            <span className="text-[9px] text-slate-500 flex items-center gap-1">
   528→              <ChevronLeft className="h-3 w-3 rotate-90" />
   529→              Scroll for more
   530→              <ChevronLeft className="h-3 w-3 -rotate-90" />
   531→            </span>
   532→          </div>
   533→        </div>
   534→      )}
   535→
   536→      {/* Footer */}
   537→      {!collapsed && (
   538→        <div className="border-t border-slate-700/50 px-4 py-3">
   539→          <div className="flex items-center gap-2 text-xs text-slate-500">
   540→            <CircleDot className="h-3 w-3 text-emerald-500" />
   541→            <span>System Online</span>
   542→            <span className="ml-auto">v4.0.0</span>
   543→          </div>
   544→        </div>
   545→      )}
   546→    </div>
   547→  );
   548→}
   549→
   550→const severityIcons: Record<string, React.ElementType> = {
   551→  CRITICAL: XCircle,
   552→  WARNING: AlertTriangle,
   553→  INFO: Info,
   554→};
   555→
   556→const severityColors: Record<string, string> = {
   557→  CRITICAL: 'text-red-500',
   558→  WARNING: 'text-amber-500',
   559→  INFO: 'text-slate-400',
   560→};
   561→
   562→export default function DmrvApp() {
   563→  const [activeModule, setActiveModule] = useState<ModuleKey>('dashboard');
   564→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   565→  const [mobileOpen, setMobileOpen] = useState(false);
   566→  const [notifications, setNotifications] = useState<Notification[]>([]);
   567→  const [systemStatus, setSystemStatus] = useState<SystemStatus | null>(null);
   568→  const [itemsWithBadges, setItemsWithBadges] = useState<MenuItem[]>(menuItems);
   569→  // Gate Radix interactive components (Sheet, Popover) to client-only render.
   570→  // Radix uses React's useId() to generate aria-controls IDs which can mismatch
   571→  // between server and client SSR renders. useSyncExternalStore returns false
   572→  // during SSR and initial hydration (matching server HTML exactly), then true
   573→  // after mount — eliminating the hydration mismatch without cascading renders.
   574→  const mounted = useSyncExternalStore(
   575→    () => () => {},
   576→    () => true,
   577→    () => false
   578→  );
   579→
   580→  const ActiveComponent = moduleComponents[activeModule];
   581→
   582→  const activeItem = menuItems.find((item) => item.key === activeModule);
   583→  const unresolvedCount = notifications.filter((n) => !n.resolved).length;
   584→
   585→  const fetchSystemData = useCallback(async () => {
   586→    try {
   587→      const [monitoringRes, submissionRes, certificationRes, conflictRes] = await Promise.all([
   588→        fetch('/api/dmrv/monitoring'),
   589→        fetch('/api/dmrv/submission'),
   590→        fetch('/api/dmrv/certification'),
   591→        fetch('/api/dmrv/conflict'),
   592→      ]);
   593→
   594→      if (monitoringRes.ok) {
   595→        const monitoringData = await monitoringRes.json();
   596→        const health = monitoringData.health as SystemStatus;
   597→        setSystemStatus(health);
   598→        setNotifications(
   599→          (monitoringData.alerts || []).map((a: { id: string; severity: string; message: string; timestamp: string; resolved: boolean }) => ({
   600→            id: a.id,
   601→            severity: a.severity,
   602→            message: a.message,
   603→            timestamp: a.timestamp,
   604→            resolved: a.resolved,
   605→          }))
   606→        );
   607→
   608→        const badges: Partial<Record<ModuleKey, { badge: string; badgeColor: string }>> = {};
   609→
   610→        if (health?.alerts?.unresolved > 0) {
   611→          badges.monitoring = { badge: String(health.alerts.unresolved), badgeColor: health.alerts.critical > 0 ? 'bg-red-600 hover:bg-red-600' : 'bg-amber-600 hover:bg-amber-600' };
   612→        }
   613→
   614→        setItemsWithBadges(
   615→          menuItems.map((item) => ({
   616→            ...item,
   617→            badge: badges[item.key]?.badge,
   618→            badgeColor: badges[item.key]?.badgeColor,
   619→          }))
   620→        );
   621→      }
   622→
   623→      if (submissionRes.ok) {
   624→        const subData = await submissionRes.json();
   625→        const pendingSubs = subData.summary?.readyForVerification || 0;
   626→        if (pendingSubs > 0) {
   627→          setItemsWithBadges((prev) =>
   628→            prev.map((item) =>
   629→              item.key === 'submission'
   630→                ? { ...item, badge: String(pendingSubs), badgeColor: 'bg-amber-600 hover:bg-amber-600' }
   631→                : item
   632→            )
   633→          );
   634→        }
   635→      }
   636→
   637→      if (certificationRes.ok) {
   638→        const certData = await certificationRes.json();
   639→        const pendingCerts = certData.summary?.submitted || 0;
   640→        if (pendingCerts > 0) {
   641→          setItemsWithBadges((prev) =>
   642→            prev.map((item) =>
   643→              item.key === 'certification'
   644→                ? { ...item, badge: String(pendingCerts), badgeColor: 'bg-teal-600 hover:bg-teal-600' }
   645→                : item
   646→            )
   647→          );
   648→        }
   649→      }
   650→
   651→      if (conflictRes.ok) {
   652→        const conflictData = await conflictRes.json();
   653→        const flaggedCases = conflictData.summary?.byStatus?.FLAGGED || 0;
   654→        const criticalCases = conflictData.summary?.bySeverity?.CRITICAL || 0;
   655→        const totalOpen = flaggedCases + (conflictData.summary?.byStatus?.IN_REVIEW || 0);
   656→        if (totalOpen > 0) {
   657→          setItemsWithBadges((prev) =>
   658→            prev.map((item) =>
   659→              item.key === 'conflict'
   660→                ? { ...item, badge: String(totalOpen), badgeColor: criticalCases > 0 ? 'bg-red-600 hover:bg-red-600' : 'bg-amber-600 hover:bg-amber-600' }
   661→                : item
   662→            )
   663→          );
   664→        }
   665→      }
   666→    } catch {
   667→      // Silently fail - the shell should still work
   668→    }
   669→  }, []);
   670→
   671→  useEffect(() => {
   672→    let cancelled = false;
   673→    const load = async () => {
   674→      if (!cancelled) await fetchSystemData();
   675→    };
   676→    load();
   677→    const interval = setInterval(() => {
   678→      if (!cancelled) fetchSystemData();
   679→    }, 60000);
   680→    return () => { cancelled = true; clearInterval(interval); };
   681→  }, [fetchSystemData]);
   682→
   683→  const handleModuleChange = (key: ModuleKey) => {
   684→    setActiveModule(key);
   685→    setMobileOpen(false);
   686→  };
   687→
   688→  const systemStatusLabel = systemStatus?.status === 'HEALTHY'
   689→    ? 'System Healthy'
   690→    : systemStatus?.status === 'WARNING'
   691→    ? 'Warnings Active'
   692→    : systemStatus?.status === 'CRITICAL'
   693→    ? 'Critical Alert'
   694→    : 'Loading...';
   695→
   696→  const statusDotColor = systemStatus?.status === 'HEALTHY'
   697→    ? 'bg-emerald-500'
   698→    : systemStatus?.status === 'WARNING'
   699→    ? 'bg-amber-500'
   700→    : systemStatus?.status === 'CRITICAL'
   701→    ? 'bg-red-500'
   702→    : 'bg-slate-400';
   703→
   704→  return (
   705→    <div className="min-h-screen flex flex-col bg-gray-50">
   706→      {/* Header */}
   707→      <header className="sticky top-0 z-40 flex h-14 items-center gap-4 border-b bg-white px-4 shadow-sm lg:px-6">
   708→        {/* Mobile menu trigger */}
   709→        {mounted ? (
   710→          <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>
   711→            <SheetTrigger asChild>
   712→              <Button variant="ghost" size="icon" className="lg:hidden">
   713→                <Menu className="h-5 w-5" />
   714→                <span className="sr-only">Toggle navigation menu</span>
   715→              </Button>
   716→            </SheetTrigger>
   717→            <SheetContent side="left" className="w-72 p-0">
   718→              <SheetTitle className="sr-only">Navigation Menu</SheetTitle>
   719→              <SidebarContent activeModule={activeModule} onModuleChange={handleModuleChange} menuItemsWithBadges={itemsWithBadges} />
   720→            </SheetContent>
   721→          </Sheet>
   722→        ) : (
   723→          <Button variant="ghost" size="icon" className="lg:hidden" aria-hidden tabIndex={-1}>
   724→            <Menu className="h-5 w-5" />
   725→          </Button>
   726→        )}
   727→
   728→        {/* Breadcrumb / Title */}
   729→        <div className="flex items-center gap-2">
   730→          {activeItem && (
   731→            <>
   732→              <activeItem.icon className="h-5 w-5 text-emerald-600" />
   733→              <h1 className="text-base font-semibold text-gray-900">{activeItem.label}</h1>
   734→              {activeItem.section === 'ENTERPRISE' && (
   735→                <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 text-[9px] h-4 ml-1">ENT</Badge>
   736→              )}
   737→            </>
   738→          )}
   739→        </div>
   740→
   741→        <div className="ml-auto flex items-center gap-3">
   742→          {/* Status Indicators */}
   743→          <div className="hidden sm:flex items-center gap-2 text-xs text-muted-foreground">
   744→            <div className={`flex items-center gap-1.5 rounded-full px-2.5 py-1 ${
   745→              systemStatus?.status === 'HEALTHY' ? 'bg-emerald-50 text-emerald-700' :
   746→              systemStatus?.status === 'WARNING' ? 'bg-amber-50 text-amber-700' :
   747→              systemStatus?.status === 'CRITICAL' ? 'bg-red-50 text-red-700' :
   748→              'bg-slate-50 text-slate-500'
   749→            }`}>
   750→              <div className={`h-1.5 w-1.5 rounded-full ${statusDotColor} animate-pulse`} />
   751→              <span>{systemStatusLabel}</span>
   752→              {systemStatus && (
   753→                <span className="text-[10px] opacity-70">
   754→                  ({systemStatus.projects.active} proj)
   755→                </span>
   756→              )}
   757→            </div>
   758→          </div>
   759→          <Separator orientation="vertical" className="h-6 hidden sm:block" />
   760→
   761→          {/* Notification Bell */}
   762→          {mounted ? (
   763→            <Popover>
   764→              <PopoverTrigger asChild>
   765→                <Button variant="ghost" size="icon" className="relative">
   766→                  <Bell className="h-4 w-4" />
   767→                  {unresolvedCount > 0 && (
   768→                    <span className={`absolute -top-0.5 -right-0.5 h-3.5 w-3.5 rounded-full text-[9px] text-white flex items-center justify-center font-bold ${
   769→                      notifications.some((n) => n.severity === 'CRITICAL' && !n.resolved) ? 'bg-red-500' : 'bg-amber-500'
   770→                    }`}>
   771→                      {unresolvedCount > 9 ? '9+' : unresolvedCount}
   772→                    </span>
   773→                  )}
   774→                </Button>
   775→              </PopoverTrigger>
   776→              <PopoverContent className="w-80 p-0" align="end">
   777→                <div className="flex items-center justify-between px-4 py-3 border-b">
   778→                  <h4 className="text-sm font-semibold">Notifications</h4>
   779→                  <Button variant="ghost" size="sm" onClick={fetchSystemData} className="h-7 gap-1 text-xs">
   780→                    <RefreshCw className="h-3 w-3" />
   781→                    Refresh
   782→                  </Button>
   783→                </div>
   784→                <ScrollArea className="max-h-80">
   785→                  {notifications.length === 0 ? (
   786→                    <div className="py-8 text-center text-sm text-muted-foreground">
   787→                      No notifications
   788→                    </div>
   789→                  ) : (
   790→                    <div className="px-4">
   791→                      {notifications.slice(0, 10).map((notif, idx) => {
   792→                        const Icon = severityIcons[notif.severity] || Info;
   793→                        return (
   794→                          <React.Fragment key={notif.id}>
   795→                            <div className={`flex items-start gap-3 py-3 ${notif.resolved ? 'opacity-50' : ''}`}>
   796→                              <Icon className={`mt-0.5 h-4 w-4 shrink-0 ${severityColors[notif.severity] || 'text-slate-400'}`} />
   797→                              <div className="flex-1 min-w-0">
   798→                                <p className="text-xs leading-relaxed">{notif.message}</p>
   799→                                <div className="flex items-center gap-2 mt-1">
   800→                                  <Badge variant="secondary" className={`text-[9px] h-4 ${
   801→                                    notif.severity === 'CRITICAL' ? 'bg-red-100 text-red-700 hover:bg-red-100' :
   802→                                    notif.severity === 'WARNING' ? 'bg-amber-100 text-amber-700 hover:bg-amber-100' :
   803→                                    'bg-slate-100 text-slate-600 hover:bg-slate-100'
   804→                                  }`}>
   805→                                    {notif.severity}
   806→                                  </Badge>
   807→                                  {notif.resolved && (
   808→                                    <Badge variant="secondary" className="text-[9px] h-4 bg-emerald-50 text-emerald-600 hover:bg-emerald-50">
   809→                                      RESOLVED
   810→                                    </Badge>
   811→                                  )}
   812→                                </div>
   813→                              </div>
   814→                            </div>
   815→                            {idx < Math.min(notifications.length, 10) - 1 && <Separator />}
   816→                          </React.Fragment>
   817→                        );
   818→                      })}
   819→                    </div>
   820→                  )}
   821→                </ScrollArea>
   822→                {unresolvedCount > 0 && (
   823→                  <div className="border-t px-4 py-2">
   824→                    <Button
   825→                      variant="ghost"
   826→                      size="sm"
   827→                      className="w-full text-xs gap-1"
   828→                      onClick={() => handleModuleChange('monitoring')}
   829→                    >
   830→                      <Activity className="h-3 w-3" />
   831→                      View All Alerts in Monitoring
   832→                    </Button>
   833→                  </div>
   834→                )}
   835→              </PopoverContent>
   836→            </Popover>
   837→          ) : (
   838→            <Button variant="ghost" size="icon" className="relative" aria-hidden tabIndex={-1}>
   839→              <Bell className="h-4 w-4" />
   840→            </Button>
   841→          )}
   842→
   843→          <div className="flex h-8 w-8 items-center justify-center rounded-full bg-emerald-100 text-emerald-700 text-xs font-bold">
   844→            AD
   845→          </div>
   846→
   847→          {/* User menu / login */}
   848→          <UserMenuButton onNavigateToUsers={() => handleModuleChange('users')} />
   849→        </div>
   850→      </header>
   851→
   852→      <div className="flex flex-1 overflow-hidden">
   853→        {/* Desktop Sidebar */}
   854→        <aside
   855→          className={`hidden lg:flex flex-col border-r border-slate-200 transition-all duration-200 ${
   856→            sidebarCollapsed ? 'w-16' : 'w-60'
   857→          }`}
   858→        >
   859→          <SidebarContent collapsed={sidebarCollapsed} activeModule={activeModule} onModuleChange={handleModuleChange} menuItemsWithBadges={itemsWithBadges} />
   860→          {/* Collapse Toggle */}
   861→          <button
   862→            onClick={() => setSidebarCollapsed(!sidebarCollapsed)}
   863→            className="flex items-center justify-center py-2 text-slate-400 hover:text-white bg-slate-900 border-t border-slate-700/50 transition-colors"
   864→            aria-label={sidebarCollapsed ? 'Expand sidebar' : 'Collapse sidebar'}
   865→          >
   866→            <ChevronLeft
   867→              className={`h-4 w-4 transition-transform ${sidebarCollapsed ? 'rotate-180' : ''}`}
   868→            />
   869→          </button>
   870→        </aside>
   871→
   872→        {/* Main Content */}
   873→        <main className={`flex-1 ${activeModule === 'map' ? 'overflow-hidden' : 'overflow-y-auto'}`}>
   874→          <div className={`${activeModule === 'map' ? 'h-full p-0' : 'p-4 lg:p-6 max-w-7xl mx-auto'}`}>
   875→            <ErrorBoundary key={activeModule}>
   876→              <React.Suspense fallback={
   877→                <div className="flex items-center justify-center h-64">
   878→                  <div className="flex flex-col items-center gap-3">
   879→                    <RefreshCw className="h-8 w-8 text-emerald-600 animate-spin" />
   880→                    <span className="text-sm text-muted-foreground">Loading module...</span>
   881→                  </div>
   882→                </div>
   883→              }>
   884→                <ActiveComponent />
   885→              </React.Suspense>
   886→            </ErrorBoundary>
   887→          </div>
   888→        </main>
   889→      </div>
   890→    </div>
   891→  );
   892→}
