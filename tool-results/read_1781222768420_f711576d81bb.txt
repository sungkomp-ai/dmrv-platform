   193→// ─── Helper: Generate Integrity Hash ──────────────────────────────────
   194→function generateHash(sensorId: string, value: number, timestamp: string): string {
   195→  // Simplified SHA-like hash for demo
   196→  const raw = `${sensorId}:${value}:${timestamp}`;
   197→  let hash = 0;
   198→  for (let i = 0; i < raw.length; i++) {
   199→    const char = raw.charCodeAt(i);
   200→    hash = ((hash << 5) - hash) + char;
   201→    hash = hash & hash;
   202→  }
   203→  return `sha256-${Math.abs(hash).toString(16).padStart(16, '0')}`;
   204→}
   205→
   206→function generateSignature(nodeId: string, hash: string): string {
   207→  const raw = `${nodeId}:${hash}`;
   208→  let sig = 0;
   209→  for (let i = 0; i < raw.length; i++) {
   210→    const char = raw.charCodeAt(i);
   211→    sig = ((sig << 5) - sig) + char;
   212→    sig = sig & sig;
   213→  }
   214→  return `sig-${Math.abs(sig).toString(16).padStart(16, '0')}`;
   215→}

   217→function validateReading(sensor: SensorConfig, value: number): 'VALIDATED' | 'ANOMALY' | 'REJECTED' {
   218→  if (value < sensor.minRange || value > sensor.maxRange) return 'REJECTED';
   219→  // Anomaly detection: if value is in top/bottom 5% of range
   220→  const range = sensor.maxRange - sensor.minRange;
   221→  const lowerBound = sensor.minRange + range * 0.02;
   222→  const upperBound = sensor.maxRange - range * 0.02;
   223→  if (value < lowerBound || value > upperBound) return 'ANOMALY';
   224→  return 'VALIDATED';
   225→}

   245→function generateSensorValue(sensorType: string, sensor: SensorConfig, tick: number): number {
   246→  const config = BASE_VALUES[sensorType] || { base: 50, variance: 10 };
   247→  const hour = (tick % 24);
   248→  const seasonalFactor = config.seasonal ? Math.sin((hour / 24) * Math.PI * 2) * 0.2 : 0;
   249→  const noise = (Math.random() - 0.5) * config.variance;
   250→  let value = config.base * (1 + seasonalFactor) + noise;
   251→  // Clamp to sensor range
   252→  value = Math.max(sensor.minRange, Math.min(sensor.maxRange, value));
   253→  return Math.round(value * 100) / 100;
   254→}

   455→// ─── Simulation Engine ────────────────────────────────────────────────
   456→function startSimulation() {
   457→  if (simInterval) clearInterval(simInterval);
   458→  simInterval = setInterval(() => {
   459→    if (!simRunning) return;
   460→    tickCounter++;
   461→
   462→    const newReadings: SensorReading[] = [];
   463→
   464→    for (const edgeNode of EDGE_NODES) {
   465→      if (edgeNode.status !== 'online') continue;
   466→
   467→      for (const sensor of edgeNode.sensors) {
   468→        if (sensor.status !== 'active') continue;
   469→
   470→        // สุ่มค่าตาม sampling rate (เพื่อไม่ให้ทุก sensor ส่งทุก tick)
   471→        if (tickCounter % Math.max(1, Math.floor(sensor.samplingRate / 10)) !== 0) continue;
   472→
   473→        const value = generateSensorValue(sensor.sensorType, sensor, tickCounter);
   474→        const timestamp = new Date().toISOString();
   475→        const qualityFlag = validateReading(sensor, value);
   476→        const integrityHash = generateHash(sensor.sensorId, value, timestamp);
   477→        const signature = generateSignature(edgeNode.nodeId, integrityHash);
   478→
   479→        const reading: SensorReading = {
   480→          id: `RD-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   481→          edgeNodeId: edgeNode.nodeId,
   482→          sensorId: sensor.sensorId,
   483→          sensorType: sensor.sensorType,
   484→          value,
   485→          unit: sensor.unit,
   486→          timestamp,
   487→          receivedAt: new Date().toISOString(),
   488→          ingestionMode: 'realtime',
   489→          qualityFlag,
   490→          integrityHash,
   491→          signature,
   492→        };
   493→
   494→        newReadings.push(reading);
   495→        edgeNode.totalReadings++;
   496→      }
   497→
   498→      // Update heartbeat
   499→      edgeNode.lastHeartbeat = new Date().toISOString();
   500→    }
   501→
   502→    if (newReadings.length > 0) {
   503→      readings.push(...newReadings);
   504→      // เก็บแค่ 2000 รายการล่าสุด
   505→      if (readings.length > 2000) readings = readings.slice(-2000);
   506→
   507→      // Broadcast
   508→      io.emit('iot:readings', newReadings);
   509→    }
   510→
   511→    // ส่ง status update
   512→    io.emit('iot:status', {
   513→      running: simRunning,
   514→      tick: tickCounter,
   515→      edgeNodes: EDGE_NODES,
   516→      totalReadings: readings.length,
   517→      batchJobs: batchJobs.length,
   518→      connectedClients: realtimeClients.size,
   519→    });
   520→
   521→  }, 2000); // ทุก 2 วินาที
   522→}

   524→// ─── HTTP REST Endpoints ──────────────────────────────────────────────
   525→// สำหรับ Batch upload ผ่าน HTTP (เผื่อ Edge Node ที่ไม่ได้ใช้ WebSocket)
   526→
   527→const CORS_HEADERS = {
   528→  'Access-Control-Allow-Origin': '*',
   529→  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   530→  'Access-Control-Allow-Headers': 'Content-Type, Authorization',
   531→};
   532→
   533→httpServer.on('request', async (req, res) => {
   534→  // CORS preflight
   535→  if (req.method === 'OPTIONS') {
   536→    res.writeHead(204, CORS_HEADERS);
   537→    res.end();
   538→    return;
   539→  }
   540→
   541→  const url = new URL(req.url || '/', `http://localhost:${PORT}`);
   542→
   543→  // GET /api/status
   544→  if (url.pathname === '/api/status' && req.method === 'GET') {
   545→    res.writeHead(200, { 'Content-Type': 'application/json', ...CORS_HEADERS });

   548→      tick: tickCounter,
   549→      edgeNodes: EDGE_NODES,

   552→      connectedClients: realtimeClients.size,
   553→    }));
   554→    return;
   555→  }
   556→
   557→  // GET /api/readings
   558→  if (url.pathname === '/api/readings' && req.method === 'GET') {
   559→    const limit = parseInt(url.searchParams.get('limit') || '100');
   560→    const edgeNodeId = url.searchParams.get('edgeNodeId');
   561→    const sensorType = url.searchParams.get('sensorType');
   562→    let filtered = readings;
   563→    if (edgeNodeId) filtered = filtered.filter(r => r.edgeNodeId === edgeNodeId);
   564→    if (sensorType) filtered = filtered.filter(r => r.sensorType === sensorType);
   565→    res.writeHead(200, { 'Content-Type': 'application/json', ...CORS_HEADERS });
   566→    res.end(JSON.stringify({ readings: filtered.slice(-limit), total: filtered.length }));
   567→    return;
   568→  }
   569→
   570→  // GET /api/edge-nodes
   571→  if (url.pathname === '/api/edge-nodes' && req.method === 'GET') {
   572→    res.writeHead(200, { 'Content-Type': 'application/json', ...CORS_HEADERS });

   575→  }
   576→
   577→  // GET /api/batch-jobs
   578→  if (url.pathname === '/api/batch-jobs' && req.method === 'GET') {
   579→    res.writeHead(200, { 'Content-Type': 'application/json', ...CORS_HEADERS });
   580→    res.end(JSON.stringify({ batchJobs }));
   581→    return;
   582→  }
   583→
   584→  // POST /api/batch-upload
   585→  if (url.pathname === '/api/batch-upload' && req.method === 'POST') {
   586→    let body = '';
   587→    req.on('data', chunk => body += chunk);
   588→    req.on('end', () => {
   589→      try {
   590→        const data = JSON.parse(body);

   592→        if (!edgeNode) {
   593→          res.writeHead(404, { 'Content-Type': 'application/json', ...CORS_HEADERS });
   594→          res.end(JSON.stringify({ error: 'Edge node not found' }));
   595→          return;
   596→        }
   597→
   598→        const batchJob: BatchJobState = {
   599→          id: `BATCH-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   600→          edgeNodeId: data.edgeNodeId,
   601→          fileName: data.fileName || 'upload.json',
   602→          fileType: data.fileType || 'JSON',
   603→          totalRecords: data.records?.length || 0,
   604→          processedRecords: 0,
   605→          validRecords: 0,
   606→          rejectedRecords: 0,
   607→          anomalyRecords: 0,
   608→          status: 'COMPLETED',
   609→          certificationStatus: 'CERTIFIED',
   610→          merkleRoot: generateHash('batch', data.records?.length || 0, new Date().toISOString()),
   611→          createdAt: new Date().toISOString(),
   612→          completedAt: new Date().toISOString(),
   613→        };
   614→
   615→        const batchReadings: SensorReading[] = (data.records || []).map((record: any) => {
   616→          const sensor = edgeNode.sensors.find(s => s.sensorId === record.sensorId);
   617→          const qualityFlag = sensor ? validateReading(sensor, record.value) : 'RAW';
   618→          const integrityHash = generateHash(record.sensorId, record.value, record.timestamp);
   619→          const signature = generateSignature(data.edgeNodeId, integrityHash);
   620→
   621→          if (qualityFlag === 'VALIDATED') batchJob.validRecords++;
   622→          else if (qualityFlag === 'ANOMALY') { batchJob.anomalyRecords++; batchJob.validRecords++; }
   623→          else batchJob.rejectedRecords++;
   624→
   625→          batchJob.processedRecords++;
   626→
   627→          return {
   628→            id: `RD-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   629→            edgeNodeId: data.edgeNodeId,
   630→            sensorId: record.sensorId,
   631→            sensorType: record.sensorType,
   632→            value: record.value,
   633→            unit: record.unit || '',
   634→            timestamp: record.timestamp,
   635→            receivedAt: new Date().toISOString(),
   636→            ingestionMode: 'batch' as const,
   637→            qualityFlag,
   638→            integrityHash,
   639→            signature,
   640→          };
   641→        });

   651→      } catch (e: any) {
   652→        res.writeHead(400, { 'Content-Type': 'application/json', ...CORS_HEADERS });
   653→        res.end(JSON.stringify({ error: e.message }));
   654→      }
   655→    });

   657→  }

   677→
   678→        const reading: SensorReading = {
   679→          id: `RD-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   680→          edgeNodeId: data.edgeNodeId,
   681→          sensorId: data.sensorId,
   682→          sensorType: data.sensorType,
   683→          value: data.value,
