   511→    const section = searchParams.get('section') || 'all'; // all, plots, measurements, calculations, curl
   512→
   513→    // ─── ถ้าต้องการ seed ข้อมูลลงฐานข้อมูล ───
   514→    if (seed) {
   515→      const seededPlots: any[] = [];
   516→      const seededTerraces: any[] = [];
   517→      const seededCheckDams: any[] = [];
   518→      const seededMeasurements: any[] = [];
   519→
   520→      for (const plotData of SAMPLE_PLOTS) {
   521→        // ตรวจสอบว่ามีอยู่แล้วหรือไม่
   522→        const existing = await db.soilCarbonPlot.findUnique({ where: { plotId: plotData.plotId } });

   524→          seededPlots.push(existing);
   525→          continue;
   526→        }
   527→
   528→        // สร้างแปลง
   529→        const plot = await db.soilCarbonPlot.create({ data: plotData });
   530→        seededPlots.push(plot);
   531→
   532→        // ─── สร้างฝายชะลอน้ำก่อน (เพื่อให้ TerraceLayer อ้างอิงได้) ───
   533→        const damIds: string[] = [];
   534→        const sampleDams = SAMPLE_CHECK_DAMS[plotData.plotId] || [];
   535→        for (const damData of sampleDams) {
   536→          const dam = await db.checkDam.create({
   537→            data: {
   538→              plotId: plot.plotId,
   539→              damId: damData.damId,
   540→              damName: `Check Dam ${damData.damId}`,
   541→              damNameTh: damData.damNameTh,

   564→        for (const terraceData of sampleTerraces) {
