   308→   308→function getGrowthStage(dap: number, totalDays: number): string {
   309→   309→  const p = dap / totalDays;
   310→   310→  if (p < 0.05) return 'GERMINATION';
   311→   311→  if (p < 0.20) return 'TILLERING';
   312→   312→  if (p < 0.40) return 'STEM_ELONGATION';
   313→   313→  if (p < 0.55) return 'BOOTING';
   314→   314→  if (p < 0.65) return 'FLOWERING';
   315→   315→  if (p < 0.85) return 'GRAIN_FILLING';
   316→   316→  if (p < 1.00) return 'MATURITY';
   317→   317→  return 'HARVESTED';
   318→   318→}

   321→   321→function getTemperature(field: AWDField, hour: number): number {
   322→   322→  // อุณหภูมิตามช่วงเวลา (รอบ 24 ชม.) sin curve
   323→   323→  const avgTemp = (field.baseTempMin + field.baseTempMax) / 2;
   324→   324→  const amplitude = (field.baseTempMax - field.baseTempMin) / 2;
   325→   325→  // peak at 14:00, low at 05:00
   326→   326→  const temp = avgTemp + amplitude * Math.sin(((hour - 5) / 24) * Math.PI * 2 - Math.PI / 2);
   327→   327→  // เพิ่ม noise เล็กน้อย
   328→   328→  return temp + (Math.sin(field.currentDAP * 0.7 + hour * 0.3) * 1.5);
   329→   329→}

   331→   331→// ─── Rainfall Model ──────────────────────────────────────────────────
   332→   332→function getRainfallMm(field: AWDField, dap: number, hour: number): number {
   333→   333→  // ฝนตกตามฤดู: มิ.ย.-ต.ค. = ฝนชุก, พ.ย.-พ.ค. = แล้ง
   334→   334→  // DAP 0-30 = ต้นฤดู (ฝนเริ่ม), 30-80 = กลางฤดู (ฝนชุก), 80+ = ปลายฤดู
   335→   335→  const rainFactor = field.annualRainfallMm / 1400;
   336→   336→  let baseRainfall = 0;
   337→   337→  if (dap < 30) baseRainfall = 0.15;       // ต้นฤดู 15% โอกาสฝนตก
   338→   338→  else if (dap < 80) baseRainfall = 0.40;  // กลางฤดู 40%
   339→   339→  else baseRainfall = 0.20;                // ปลายฤดู 20%
   340→   340→
   341→   341→  // ฝนตกเฉพาะบางช่วงเวลา (12:00-18:00)
   342→   342→  const isRainyHour = hour >= 12 && hour <= 18;
   343→   343→  if (!isRainyHour) return 0;
   344→   344→
   345→   345→  // ใช้ deterministic pseudo-random จาก DAP + hour
   346→   346→  const seed = Math.sin(dap * 12.9898 + hour * 78.233) * 43758.5453;
   347→   347→  const rand = seed - Math.floor(seed);
   348→   348→  if (rand > baseRainfall * rainFactor) return 0;
   349→   349→
   350→   350→  // ปริมาณฝน 0.5-8 mm/hr
   351→   351→  const intensity = (rand * 7.5 + 0.5) * rainFactor;
   352→   352→  return Math.round(intensity * 10) / 10;
   353→   353→}

   355→   355→// ─── CH4 Emission Model (T-VER-P-AG-001) ────────────────────────────
   356→   356→// CH4 ปล่อยมากตอนน้ำท่วม ลดลงมากตอนแห้ง (AWD ช่วยลด CH4)
   357→   357→// หน่วย: mg/m²/hr → แปลงเป็น kg CH₄/ha/day สำหรับ T-VER
   358→   358→function getCH4Emission(field: AWDField, hour: number): number {
   359→   359→  const stage = getGrowthStage(field.currentDAP, field.totalDays);
   360→   360→  // อัตราปล่อยพื้นฐานตามช่วงเจริญเติบโต (mg/m²/hr)
   361→   361→  // อ้างอิง: T-VER-P-AG-001 EF scaling by growth stage
   362→   362→  let baseFlux = 0;
   363→   363→  switch (stage) {
   364→   364→    case 'GERMINATION': baseFlux = 2; break;
   365→   365→    case 'TILLERING': baseFlux = 15; break;
   366→   366→    case 'STEM_ELONGATION': baseFlux = 25; break;
   367→   367→    case 'BOOTING': baseFlux = 30; break;
   368→   368→    case 'FLOWERING': baseFlux = 28; break;
   369→   369→    case 'GRAIN_FILLING': baseFlux = 18; break;
   370→   370→    case 'MATURITY': baseFlux = 8; break;
   371→   371→    case 'HARVESTED': baseFlux = 1; break;
   372→   372→  }
   373→   373→
   374→   374→  // ปัจจัยระดับน้ำ
   375→   375→  let waterFactor = 1.0;
   376→   376→  switch (field.currentAWDPhase) {
   377→   377→    case 'FLOODED': waterFactor = 1.0; break;
   378→   378→    case 'DRAINING': waterFactor = 0.6; break;
   379→   379→    case 'DRY': waterFactor = 0.15; break; // ลดมากตอนแห้ง
   380→   380→    case 'REFLOODING': waterFactor = 0.7; break;
   381→   381→  }
   382→   382→
   383→   383→  // อุณหภูมิ affect: สูงขึ้น = CH4 เพิ่ม
   384→   384→  const temp = getTemperature(field, hour);
   385→   385→  const tempFactor = 1 + (temp - 28) * 0.03;
   386→   386→
   387→   387→  // ดินเหนียว = ปล่อย CH4 มากกว่าดินทราย
   388→   388→  const soilFactor = field.soilType.includes('Clay') ? 1.3 :
   389→   389→                     field.soilType.includes('Sandy') ? 0.7 : 1.0;
   390→   390→
   391→   391→  return Math.max(0, baseFlux * waterFactor * tempFactor * soilFactor);
   392→   392→}
   393→   393→
   394→   394→// ─── N2O Emission Model (T-VER-P-AG-001) ────────────────────────────
   395→   395→// N2O เพิ่มตอน drain/reflood (nitrification/denitrification)
   396→   396→function getN2OEmission(field: AWDField): number {
   397→   397→  const stage = getGrowthStage(field.currentDAP, field.totalDays);
   398→   398→  let baseFlux = 0;
   399→   399→  switch (stage) {
   400→   400→    case 'GERMINATION': baseFlux = 0.3; break;
   401→   401→    case 'TILLERING': baseFlux = 1.2; break;
   402→   402→    case 'STEM_ELONGATION': baseFlux = 1.5; break;
   403→   403→    case 'BOOTING': baseFlux = 1.8; break;
   404→   404→    case 'FLOWERING': baseFlux = 1.5; break;
   405→   405→    case 'GRAIN_FILLING': baseFlux = 1.0; break;
   406→   406→    case 'MATURITY': baseFlux = 0.5; break;
   407→   407→    case 'HARVESTED': baseFlux = 0.1; break;
   408→   408→  }
   409→   409→
   410→   410→  // N2O spike ตอน drain และ reflood
   411→   411→  let phaseFactor = 1.0;
   412→   412→  switch (field.currentAWDPhase) {
   413→   413→    case 'FLOODED': phaseFactor = 0.5; break;
   414→   414→    case 'DRAINING': phaseFactor = 2.5; break;  // spike!
   415→   415→    case 'DRY': phaseFactor = 1.2; break;
   416→   416→    case 'REFLOODING': phaseFactor = 2.8; break; // spike!
   417→   417→  }
   418→   418→
   419→   419→  // ดินทราย = N2O มากกว่า (leaching)
   420→   420→  const soilFactor = field.soilType.includes('Sandy') ? 1.4 :
   421→   421→                     field.soilType.includes('Clay') ? 0.8 : 1.0;
   422→   422→
   423→   423→  return Math.max(0, baseFlux * phaseFactor * soilFactor);
   424→   424→}

   426→   426→// ─── Water Level Update ──────────────────────────────────────────────
   427→   427→function updateWaterLevel(field: AWDField, rainfall: number): void {
   428→   428→  const hour = field.currentHour;
   429→   429→
   430→   430→  // การระเหย (ET) ขึ้นกับอุณหภูมิและช่วงเวลา
   431→   431→  const temp = getTemperature(field, hour);
   432→   432→  const dayETmm = 3 + (temp - 25) * 0.15; // mm/day
   433→   433→  const hourETmm = dayETmm / 24;
   434→   434→  // percolation ขึ้นกับชนิดดิน
   435→   435→  const percMmPerDay = field.soilType.includes('Sandy') ? 8 :
   436→   436→                       field.soilType.includes('Clay') ? 1.5 : 3.5;
   437→   437→  const percMmPerHr = percMmPerDay / 24;
   438→   438→
   439→   439→  // น้ำเข้า: ฝน
   440→   440→  const inflowMm = rainfall;
   441→   441→  // น้ำออก: ระเหย + ซึม
   442→   442→  const outflowMm = hourETmm + percMmPerHr;
   443→   443→
   444→   444→  // แปลง mm → cm
   445→   445→  const waterChangeCm = (inflowMm - outflowMm) / 10;
   446→   446→
   447→   447→  // ชลประทาน: เติมน้ำตอน FLOODED/REFLOODING
   448→   448→  let irrigationCm = 0;
   449→   449→  if (field.waterSource.includes('ชลประทาน') &&
   450→   450→      (field.currentAWDPhase === 'FLOODED' || field.currentAWDPhase === 'REFLOODING') &&
   451→   451→      hour === 7) { // เติมน้ำตอนเช้า
   452→   452→    irrigationCm = 0.5; // cm/hr เมื่อเปิดประตูน้ำ
   453→   453→  }
   454→   454→  // สูบน้ำสำรอง (สุพรรณ)
   455→   455→  if (field.waterSource.includes('สำรอง') &&
   456→   456→      field.currentAWDPhase === 'FLOODED' && field.waterLevelCm < 3 &&
   457→   457→      hour === 8) {
   458→   458→    irrigationCm = 0.3;
   459→   459→  }
   460→   460→
   461→   461→  field.waterLevelCm += waterChangeCm + irrigationCm;
   462→   462→
   463→   463→  // AWD Phase transitions
   464→   464→  field.awdPhaseDayCounter += 1 / 24; // เพิ่ม 1 ชม. เป็นเศษของวัน
   465→   465→
   466→   466→  switch (field.currentAWDPhase) {
   467→   467→    case 'FLOODED':
   468→   468→      if (field.awdPhaseDayCounter >= field.floodDays) {
   469→   469→        field.currentAWDPhase = 'DRAINING';
   470→   470→        field.awdPhaseDayCounter = 0;
   471→   471→      }
   472→   472→      // รักษาระดับน้ำถ้าต่ำเกิน
   473→   473→      if (field.waterLevelCm < 2 && field.waterSource.includes('ชลประทาน')) {
   474→   474→        field.waterLevelCm = Math.max(field.waterLevelCm, 3);
   475→   475→      }
   476→   476→      break;
   477→   477→
   478→   478→    case 'DRAINING':
   479→   479→      if (field.waterLevelCm <= field.dryThresholdCm) {
   480→   480→        field.currentAWDPhase = 'DRY';
   481→   481→        field.awdPhaseDayCounter = 0;
   482→   482→      }
   483→   483→      break;
   484→   484→
   485→   485→    case 'DRY':
   486→   486→      if (field.awdPhaseDayCounter >= field.dryDays) {
   487→   487→        field.currentAWDPhase = 'REFLOODING';
   488→   488→        field.awdPhaseDayCounter = 0;
   489→   489→      }
   490→   490→      // ฝนตกหนักอาจเติมน้ำให้เร็วขึ้น
   491→   491→      if (rainfall > 5 && field.waterLevelCm > 0) {
   492→   492→        field.currentAWDPhase = 'REFLOODING';
   493→   493→        field.awdPhaseDayCounter = 0;
   494→   494→      }
   495→   495→      break;
   496→   496→
   497→   497→    case 'REFLOODING':
   498→   498→      if (field.waterLevelCm >= field.floodTargetCm) {
   499→   499→        field.currentAWDPhase = 'FLOODED';
   500→   500→        field.awdPhaseDayCounter = 0;
   501→   501→        field.totalAWDCycles++;
   502→   502→      }
   503→   503→      // ชลประทานช่วยเติมน้ำ
   504→   504→      if (field.waterSource.includes('ชลประทาน') && field.waterLevelCm < 5) {
   505→   505→        field.waterLevelCm += 0.3;
   506→   506→      }
   507→   507→      break;
   508→   508→  }
   509→   509→
   510→   510→  // Clamp water level
   511→   511→  field.waterLevelCm = Math.max(-25, Math.min(20, field.waterLevelCm));
   512→   512→}

   514→   514→// ─── Soil Moisture ───────────────────────────────────────────────────
   515→   515→function getSoilMoisture(field: AWDField): number {
   516→   516→  let base = 40;
   517→   517→  switch (field.currentAWDPhase) {
   518→   518→    case 'FLOODED': base = 85 + field.waterLevelCm; break;
   519→   519→    case 'DRAINING': base = 70 + field.waterLevelCm * 0.8; break;
   520→   520→    case 'DRY': base = 30 + field.waterLevelCm * 0.5; break;
   521→   521→    case 'REFLOODING': base = 50 + field.waterLevelCm; break;
   522→   522→  }
   523→   523→  // ดินทรายแห้งเร็วกว่า
   524→   524→  if (field.soilType.includes('Sandy')) base -= 8;
   525→   525→  if (field.soilType.includes('Clay')) base += 5;
   526→   526→  return Math.max(15, Math.min(100, base));
   527→   527→}

   529→   529→// ─── Rice Yield Estimation ───────────────────────────────────────────
   530→   530→function estimateYieldKgPerHa(field: AWDField): number {
   531→   531→  // ประมาณผลผลิตจาก DAP
   532→   532→  const p = Math.min(field.currentDAP / field.totalDays, 1);
   533→   533→  const maxKgPerHa = field.riceVariety.includes('KDML') ? 3500 :
   534→   534→                     field.riceVariety.includes('RD43') ? 4200 : 4500;
   535→   535→  // S-curve growth
   536→   536→  const growthFactor = 1 / (1 + Math.exp(-10 * (p - 0.5)));
   537→   537→  // AWD ลดผลผลิตเล็กน้อย 0-5% แต่ประหยัดน้ำ 30%
   538→   538→  const awdPenalty = 0.97;
   539→   539→  // นาแล้งผลผลิตต่ำกว่า
   540→   540→  const droughtFactor = field.waterSource.includes('ฝนเท่านั้น') ? 0.75 :
   541→   541→                        field.waterSource.includes('ฝน') ? 0.88 : 1.0;
   542→   542→  return maxKgPerHa * growthFactor * awdPenalty * droughtFactor;
   543→   543→}

   545→   545→// ─── CO2 from Soil Respiration ───────────────────────────────────────
   546→   546→function getCO2Flux(field: AWDField, hour: number): number {
   547→   547→  const temp = getTemperature(field, hour);
   548→   548→  const stage = getGrowthStage(field.currentDAP, field.totalDays);
   549→   549→  let baseFlux = 50; // mg/m²/hr พื้นฐาน
   550→   550→  switch (stage) {
   551→   551→    case 'TILLERING': baseFlux = 120; break;
   552→   552→    case 'STEM_ELONGATION': baseFlux = 150; break;
   553→   553→    case 'BOOTING': baseFlux = 160; break;
   554→   554→    case 'FLOWERING': baseFlux = 140; break;
   555→   555→    case 'GRAIN_FILLING': baseFlux = 110; break;
   556→   556→    default: baseFlux = 60;
   557→   557→  }
   558→   558→  // อุณหภูมิ
   559→   559→  const tempFactor = Math.exp(0.05 * (temp - 25));
   560→   560→  // ความชื้นดิน
   561→   561→  const moisture = getSoilMoisture(field);
   562→   562→  const moistureFactor = moisture > 60 ? 1.2 : moisture > 30 ? 1.0 : 0.7;
   563→   563→  return baseFlux * tempFactor * moistureFactor;
   564→   564→}

   567→   567→const httpServer = createServer();
   568→   568→const io = new Server(httpServer, { cors: { origin: '*' } });

   576→   576→function simTick() {
   577→   577→  simTickCounter++;
   578→   578→  const newReadings: any[] = [];
   579→   579→
   580→   580→  for (const field of fields) {
   581→   581→    if (field.seasonComplete) continue;
   582→   582→
   583→   583→    // Advance time
   584→   584→    field.currentHour += SIM_HOURS_PER_TICK;
   585→   585→    if (field.currentHour >= 24) {
   586→   586→      field.currentHour -= 24;
   587→   587→      field.currentDAP++;
   588→   588→    }
   589→   589→
   590→   590→    // Check season end
   591→   591→    if (field.currentDAP >= field.totalDays) {
   592→   592→      field.seasonComplete = true;
   593→   593→      field.currentAWDPhase = 'DRY';
   594→   594→      field.waterLevelCm = -20;
   595→   595→    }
   596→   596→
   597→   597→    // Environment
   598→   598→    const temp = getTemperature(field, field.currentHour);
   599→   599→    const humidity = field.baseHumidity +
   600→   600→      (field.currentAWDPhase === 'DRY' ? -8 : field.currentAWDPhase === 'FLOODED' ? 5 : 0) +
   601→   601→      Math.sin(field.currentDAP * 0.5 + field.currentHour * 0.2) * 5;
   602→   602→    const rainfall = getRainfallMm(field, field.currentDAP, field.currentHour);
   603→   603→
   604→   604→    // Water
   605→   605→    updateWaterLevel(field, rainfall);
   606→   606→
   607→   607→    // Emissions
   608→   608→    const ch4Flux = getCH4Emission(field, field.currentHour);
   609→   609→    const n2oFlux = getN2OEmission(field);
   610→   610→    const co2Flux = getCO2Flux(field, field.currentHour);
   611→   611→
   612→   612→    // Soil
   613→   613→    const soilMoisture = getSoilMoisture(field);
   614→   614→    const soilPH = 5.5 + Math.sin(field.currentDAP * 0.1) * 0.3;
   615→   615→    const soilTempC = temp - 2 + Math.sin(field.currentDAP * 0.3) * 0.5;
   616→   616→
   617→   617→    // Growth
   618→   618→    const growthStage = getGrowthStage(field.currentDAP, field.totalDays);
   619→   619→    const yieldKgPerHa = estimateYieldKgPerHa(field);
   620→   620→
   621→   621→    // Accumulate — T-VER standard units
   622→   622→    const areaM2 = field.areaHa * 10000;
   623→   623→    // Convert flux mg/m²/hr → kg/ha for this hour
   624→   624→    const ch4KgPerHaThisHour = ch4Flux * MG_M2_HR_TO_KG_HA_DAY / 24; // kg CH₄/ha for 1 hr
   625→   625→    const n2oKgPerHaThisHour = n2oFlux * MG_M2_HR_TO_KG_HA_DAY / 24; // kg N₂O/ha for 1 hr
   626→   626→    // Total kg = per ha × area
   627→   627→    const ch4KgThisHour = ch4KgPerHaThisHour * field.areaHa;
   628→   628→    const n2oKgThisHour = n2oKgPerHaThisHour * field.areaHa;
   629→   629→
   630→   630→    field.totalCH4kg += ch4KgThisHour;
   631→   631→    field.totalN2Okg += n2oKgThisHour;
   632→   632→    field.totalCH4tCO2e = field.totalCH4kg * KG_CH4_TO_TCO2E;
   633→   633→    field.totalN2OtCO2e = field.totalN2Okg * KG_N2O_TO_TCO2E;
   634→   634→
   635→   635→    // CO₂ soil respiration offset (plant uptake ~offsets soil respiration)
   636→   636→    const co2KgThisHour = co2Flux * areaM2 / 1e6; // mg/m²/hr → kg
   637→   637→    field.totalNettCO2e = field.totalCH4tCO2e + field.totalN2OtCO2e - (co2KgThisHour / 1000 * 0.3); // 30% offset
   638→   638→    field.totalYieldKg = yieldKgPerHa * field.areaHa;
   639→   639→
   640→   640→    // T-VER-P-AG-001: Baseline (continuous flooding) emission calculation
   641→   641→    // Baseline EF depends on water source type
   642→   642→    const baselineEF = field.waterSource.includes('ชลประทาน') ? TVER_EF_CF_IRRIGATED :
   643→   643→                       field.waterSource.includes('ฝนเท่านั้น') ? TVER_EF_CF_DRY :
   644→   644→                       TVER_EF_CF_RAINFED;
   645→   645→    const baselineCH4KgPerDay = baselineEF * field.areaHa; // kg CH₄/day for whole area
   646→   646→    const baselineCH4KgThisHour = baselineCH4KgPerDay / 24;
   647→   647→    field.baselineEmissiontCO2e += baselineCH4KgThisHour * KG_CH4_TO_TCO2E;
   648→   648→    field.emissionReductiontCO2e = Math.max(0, field.baselineEmissiontCO2e - field.totalNettCO2e);
   649→   649→
   650→   650→    // Convert CH4 to CO2e (GWP=28 per TGO/IPCC AR5), N2O to CO2e (GWP=265)
   651→   651→    const ch4CO2e = ch4Flux * GWP_CH4; // mgCO2e/m²/hr
   652→   652→    const n2oCO2e = n2oFlux * GWP_N2O;
   653→   653→    const totalGHGCO2e = ch4CO2e + n2oCO2e - co2Flux; // net
   654→   654→    // AWD emission reduction vs continuous flooding
   655→   655→    const baselineCH4Flux = ch4Flux / 0.6; // continuous flooding = higher
   656→   656→    const emissionReductionPct = ((baselineCH4Flux - ch4Flux) / baselineCH4Flux) * 100;
   657→   657→
   658→   658→    const reading = {
   659→   659→      tick: simTickCounter,
   660→   660→      fieldId: field.id,
   661→   661→      fieldName: field.name,
   662→   662→      province: field.province,
   663→   663→      timestamp: new Date().toISOString(),
   664→   664→      simDAP: field.currentDAP,
   665→   665→      simHour: field.currentHour,
   666→   666→      growthStage,
   667→   667→      awdPhase: field.currentAWDPhase,
   668→   668→      awdCycles: field.totalAWDCycles,
   669→   669→      // Water
   670→   670→      waterLevelCm: Math.round(field.waterLevelCm * 10) / 10,
   671→   671→      soilMoisturePct: Math.round(soilMoisture * 10) / 10,
   672→   672→      rainfallMmHr: Math.round(rainfall * 10) / 10,
   673→   673→      // Climate
   674→   674→      airTempC: Math.round(temp * 10) / 10,
   675→   675→      soilTempC: Math.round(soilTempC * 10) / 10,
   676→   676→      humidityPct: Math.round(humidity * 10) / 10,
   677→   677→      soilPH: Math.round(soilPH * 100) / 100,
   678→   678→      // Emissions — T-VER standard units
   679→   679→      ch4FluxMgM2Hr: Math.round(ch4Flux * 100) / 100,         // mg/m²/hr (raw sensor)
   680→   680→      ch4EfKgHaDay: Math.round(ch4Flux * MG_M2_HR_TO_KG_HA_DAY * 1000) / 1000,  // kg CH₄/ha/day (T-VER EF)
   681→   681→      n2oFluxMgM2Hr: Math.round(n2oFlux * 100) / 100,
   682→   682→      n2oEfKgHaDay: Math.round(n2oFlux * MG_M2_HR_TO_KG_HA_DAY * 1000) / 1000,
   683→   683→      co2FluxMgM2Hr: Math.round(co2Flux * 10) / 10,
   684→   684→      ch4CO2eMgM2Hr: Math.round(ch4CO2e * 100) / 100,
   685→   685→      n2oCO2eMgM2Hr: Math.round(n2oCO2e * 100) / 100,
   686→   686→      netGHGCO2eMgM2Hr: Math.round(totalGHGCO2e * 100) / 100,
   687→   687→      emissionReductionPct: Math.round(emissionReductionPct * 10) / 10,
   688→   688→      // Cumulative — T-VER units
   689→   689→      cumulativeCH4kg: Math.round(field.totalCH4kg * 1000) / 1000,   // kg CH₄
   690→   690→      cumulativeN2Okg: Math.round(field.totalN2Okg * 1000) / 1000,   // kg N₂O
   691→   691→      cumulativeCH4tCO2e: Math.round(field.totalCH4tCO2e * 10000) / 10000,  // tCO₂e
   692→   692→      cumulativeN2OtCO2e: Math.round(field.totalN2OtCO2e * 10000) / 10000,
   693→   693→      cumulativeNettCO2e: Math.round(field.totalNettCO2e * 10000) / 10000,
   694→   694→      baselineEmissiontCO2e: Math.round(field.baselineEmissiontCO2e * 10000) / 10000,
   695→   695→      emissionReductiontCO2e: Math.round(field.emissionReductiontCO2e * 10000) / 10000,
   696→   696→      estimatedYieldKg: Math.round(field.totalYieldKg),
   697→   697→      seasonComplete: field.seasonComplete,
   698→   698→      // T-VER methodology reference
   699→   699→      tverMethodology: 'T-VER-P-AG-001',
   700→   700→      tverGwpCH4: GWP_CH4,
   701→   701→      tverGwpN2O: GWP_N2O,
   702→   702→    };
   703→   703→
   704→   704→    newReadings.push(reading);
   705→   705→  }
   706→   706→
   707→   707→  // Store readings
   708→   708→  readings.push(...newReadings);
   709→   709→  if (readings.length > 600) readings = readings.slice(-600);
   710→   710→
   711→   711→  // Emit via Socket.IO (local dashboard)
   712→   712→  io.emit('awd:reading', newReadings);
   713→   713→  io.emit('awd:status', getStatus());
   714→   714→
   715→   715→  // Publish via MQTT to external broker (35.213.186.240:1883)
   716→   716→  // Auto-reconnect if connection dropped mid-simulation
   717→   717→  if (!mqttConnected && !mqttClient) {
   718→   718→    connectMQTT();
   719→   719→  }
   720→   720→  for (const reading of newReadings) {
   721→   721→    publishReadingToMQTT(reading);
   722→   722→  }
   723→   723→}

   725→   725→function getStatus() {
   726→   726→  return {

   762→   762→function startSim() {
   763→   763→  if (simRunning) return;
   764→   764→  simRunning = true;
   765→   765→  // Ensure MQTT is connected to 35.213.186.240:1883 before publishing
   766→   766→  // (auto-reconnect if connection was lost or never established)
   767→   767→  if (!mqttConnected) {
   768→   768→    console.log('[SIM] MQTT not connected — reconnecting to 35.213.186.240:1883 before starting simulation...');
   769→   769→    if (mqttClient) {
   770→   770→      try { mqttClient.end(true); } catch (e) {}
   771→   771→      mqttClient = null;
   772→   772→      mqttConnected = false;
   773→   773→    }
   774→   774→    connectMQTT();
   775→   775→  }
   776→   776→  simTimer = setInterval(simTick, INTERVAL_MS);
   777→   777→  io.emit('awd:started', { speedFactor: SPEED_FACTOR });
   778→   778→  publishStatusToMQTT(getStatus());
   779→   779→}

   781→   781→function stopSim() {
   782→   782→  if (!simRunning) return;
   783→   783→  simRunning = false;
   784→   784→  if (simTimer) { clearInterval(simTimer); simTimer = null; }
   785→   785→  io.emit('awd:stopped');
   786→   786→  publishStatusToMQTT(getStatus());
   787→   787→}
