   306→// ─── Terrace Cross-Section SVG Component ──────────────────────────
   307→function TerraceCrossSection({ layers, checkDams, compact = false }: { layers: TerraceLayer[]; checkDams: CheckDam[]; compact?: boolean }) {
   308→  const h = compact ? 180 : 280;
   309→  const w = compact ? 520 : 720;
   310→  const layerCount = layers.length || 5;
   311→  const stepW = w / (layerCount + 0.5);
   312→  const stepH = h / (layerCount + 1.2);
   313→
   314→  return (
   315→    <svg width="100%" viewBox={`0 0 ${w} ${h}`} className="rounded-lg" style={{ maxHeight: compact ? 180 : 280 }}>
   316→      <defs>
   317→        <linearGradient id="terraceGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   318→          <stop offset="0%" stopColor="#065f46" />
   319→          <stop offset="100%" stopColor="#10b981" />
   320→        </linearGradient>
   321→        <linearGradient id="plantGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   322→          <stop offset="0%" stopColor="#059669" />
   323→          <stop offset="100%" stopColor="#10b981" />
   324→        </linearGradient>
   325→        <linearGradient id="ridgeGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   326→          <stop offset="0%" stopColor="#064e3b" />
   327→          <stop offset="100%" stopColor="#065f46" />
   328→        </linearGradient>
   329→        <linearGradient id="waterGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   330→          <stop offset="0%" stopColor="#3b82f6" stopOpacity="0.6" />
   331→          <stop offset="100%" stopColor="#1d4ed8" stopOpacity="0.3" />
   332→        </linearGradient>
   333→        <linearGradient id="sedimentGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   334→          <stop offset="0%" stopColor="#92400e" stopOpacity="0.8" />
   335→          <stop offset="100%" stopColor="#78350f" stopOpacity="0.6" />
   336→        </linearGradient>
   337→      </defs>
   338→
   339→      {/* Background hillside */}
   340→      <path
   341→        d={`M 0,${h * 0.15} L ${w * 0.1},${h * 0.1} L ${w},${h * 0.85} L ${w},${h} L 0,${h} Z`}
   342→        fill="#1e293b"
   343→        stroke="#334155"
   344→        strokeWidth="1"
   345→      />
   346→
   347→      {/* Terrace steps */}
   348→      {layers.map((layer, i) => {
   349→        const x1 = i * stepW * 0.7 + stepW * 0.3;
   350→        const y1 = h * 0.1 + i * stepH * 0.7;
   351→        const totalW = stepW * 1.2;
   352→        const ridgeFraction = (layer.ridgeWidthM || 0.85) / (layer.widthM || 2.35);
   353→        const plantingW = totalW * (1 - ridgeFraction);
   354→        const ridgeW = totalW * ridgeFraction;
   355→        const y2 = y1;
   356→        const y3 = y1 + stepH * 0.7;
   357→
   358→        // Planting area x range
   359→        const px1 = x1;
   360→        const px2 = x1 + plantingW;
   361→        // Ridge area x range
   362→        const rx1 = px2;
   363→        const rx2 = x1 + totalW;
   364→
   365→        const vetiverStatus = layer.vetiverGrassStatus || 'NONE';
   366→
   367→        return (
   368→          <g key={layer.id || i}>
   369→            {/* Planting area (main green) */}
   370→            <path
   371→              d={`M ${px1},${y1} L ${px2},${y2} L ${px2},${y3} L ${px1},${y3} Z`}
   372→              fill={TERRACE_COLORS[i % TERRACE_COLORS.length]}
   373→              stroke="#065f46"
   374→              strokeWidth="1"
   375→              opacity="0.85"
   376→            />
   377→
   378→            {/* Ridge area (darker green, slightly elevated) */}
   379→            <path
   380→              d={`M ${rx1},${y1 - 3} L ${rx2},${y2 - 3} L ${rx2},${y3} L ${rx1},${y3} Z`}
   381→              fill="url(#ridgeGrad)"
   382→              stroke="#064e3b"
   383→              strokeWidth="1"
   384→              opacity="0.9"
   385→            />
   386→
   387→            {/* Sediment layer */}
   388→            {layer.sedimentDepthCm > 0 && (
   389→              <rect
   390→                x={x1 + 4}
   391→                y={y2 - Math.min(layer.sedimentDepthCm * 0.6, stepH * 0.25)}
   392→                width={plantingW - 8}
   393→                height={Math.min(layer.sedimentDepthCm * 0.6, stepH * 0.25)}
   394→                fill="url(#sedimentGrad)"
   395→                rx="2"
   396→              />
   397→            )}
   398→
   399→            {/* Vetiver grass lines on ridge */}
   400→            {vetiverStatus !== 'NONE' && (
   401→              <>
   402→                {Array.from({ length: Math.max(3, Math.floor(ridgeW / 8)) }).map((_, gi) => {
   403→                  const gx = rx1 + (gi + 1) * (ridgeW / (Math.max(3, Math.floor(ridgeW / 8)) + 1));
   404→                  const grassH = vetiverStatus === 'MATURE' ? 12 : vetiverStatus === 'ESTABLISHED' ? 9 : 6;
   405→                  const grassColor = vetiverStatus === 'MATURE' ? '#34d399' : vetiverStatus === 'ESTABLISHED' ? '#6ee7b7' : '#a7f3d0';
   406→                  return (
   407→                    <g key={gi}>
   408→                      <line x1={gx - 2} y1={y1 - 3} x2={gx - 3} y2={y1 - 3 - grassH} stroke={grassColor} strokeWidth="1.2" opacity="0.8" />
   409→                      <line x1={gx} y1={y1 - 3} x2={gx + 1} y2={y1 - 3 - grassH - 2} stroke={grassColor} strokeWidth="1.2" opacity="0.9" />
   410→                      <line x1={gx + 2} y1={y1 - 3} x2={gx + 4} y2={y1 - 3 - grassH + 1} stroke={grassColor} strokeWidth="1.2" opacity="0.7" />
   411→                    </g>
   412→                  );
   413→                })}
   414→              </>
   415→            )}
   416→
   417→            {/* Check dam indicator */}
   418→            {checkDams.filter(d => d.terraceLayerId === layer.id).length > 0 && (
   419→              <rect
   420→                x={rx2 - 5}
   421→                y={y2 - 3}
   422→                width={7}
   423→                height={y3 - y2 + 3}
   424→                fill="#f59e0b"
   425→                stroke="#b45309"
   426→                strokeWidth="1"
   427→                rx="1"
   428→              />
   429→            )}
   430→
   431→            {/* Water flow arrow */}
   432→            {i < layerCount - 1 && (
   433→              <>
   434→                <line
   435→                  x1={rx2 - stepW * 0.15}
   436→                  y1={y2 + 4}
   437→                  x2={rx1 + stepW * 0.1}
   438→                  y2={y3 + stepH * 0.35}
   439→                  stroke="#3b82f6"
   440→                  strokeWidth="2"
   441→                  strokeDasharray="4,2"
   442→                  opacity="0.7"
   443→                />
   444→                <polygon
   445→                  points={`${rx1 + stepW * 0.1},${y3 + stepH * 0.35} ${rx1 + stepW * 0.02},${y3 + stepH * 0.2} ${rx1 + stepW * 0.15},${y3 + stepH * 0.25}`}
   446→                  fill="#3b82f6"
   447→                  opacity="0.7"
   448→                />
   449→              </>
   450→            )}
   451→
   452→            {/* Layer label (planting area) */}
