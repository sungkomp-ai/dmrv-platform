   132→function truncateHash(hash: string, chars = 8): string {
   133→  if (!hash || hash.length <= chars * 2 + 3) return hash;
   134→  return `${hash.slice(0, chars)}...${hash.slice(-chars)}`;
   135→}
   136→
   137→function truncateHashShort(hash: string): string {
   138→  if (!hash || hash.length <= 16) return hash;
   139→  return `${hash.slice(0, 8)}...${hash.slice(-4)}`;
   140→}
   141→
   142→function formatRelativeTime(dateStr: string): string {
   143→  const now = new Date();
   144→  const date = new Date(dateStr);
   145→  const diffMs = now.getTime() - date.getTime();
   146→  const diffSecs = Math.floor(diffMs / 1000);
   147→  const diffMins = Math.floor(diffSecs / 60);
   148→  const diffHours = Math.floor(diffMins / 60);
   149→  const diffDays = Math.floor(diffHours / 24);
   150→
   151→  if (diffSecs < 60) return `${diffSecs}s ago`;
   152→  if (diffMins < 60) return `${diffMins}m ago`;
   153→  if (diffHours < 24) return `${diffHours}h ago`;
   154→  if (diffDays < 30) return `${diffDays}d ago`;
   155→  return date.toLocaleDateString();
   156→}
   157→
   158→function getActorFromMetadata(metadataStr: string): string {
   159→  try {
   160→    const meta = JSON.parse(metadataStr) as { source?: string };
   161→    return meta.source || 'System';
   162→  } catch {
   163→    return 'System';
   164→  }
   165→}
