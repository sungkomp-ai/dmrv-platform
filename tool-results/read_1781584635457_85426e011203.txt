   270→// ─── Terrace Cross-Section SVG Component ──────────────────────────
   271→function TerraceCrossSection({ layers, checkDams, compact = false }: { layers: TerraceLayer[]; checkDams: CheckDam[]; compact?: boolean }) {
   272→  const h = compact ? 160 : 220;
   273→  const w = compact ? 480 : 640;
   274→  const layerCount = layers.length || 5;
   275→  const stepW = w / (layerCount + 0.5);
   276→  const stepH = h / (layerCount + 1.5);
   277→
   278→  return (
   279→    <svg width="100%" viewBox={`0 0 ${w} ${h}`} className="rounded-lg" style={{ maxHeight: compact ? 160 : 220 }}>
   280→      <defs>
   281→        <linearGradient id="terraceGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   282→          <stop offset="0%" stopColor="#065f46" />
   283→          <stop offset="100%" stopColor="#10b981" />
   284→        </linearGradient>
   285→        <linearGradient id="waterGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   286→          <stop offset="0%" stopColor="#3b82f6" stopOpacity="0.6" />
   287→          <stop offset="100%" stopColor="#1d4ed8" stopOpacity="0.3" />
   288→        </linearGradient>
   289→        <linearGradient id="sedimentGrad" x1="0%" y1="0%" x2="0%" y2="100%">
   290→          <stop offset="0%" stopColor="#92400e" stopOpacity="0.8" />
   291→          <stop offset="100%" stopColor="#78350f" stopOpacity="0.6" />
   292→        </linearGradient>
   293→      </defs>
   294→
   295→      {/* Background hillside */}
   296→      <path
   297→        d={`M 0,${h * 0.15} L ${w * 0.1},${h * 0.1} L ${w},${h * 0.85} L ${w},${h} L 0,${h} Z`}
   298→        fill="#1e293b"
   299→        stroke="#334155"
   300→        strokeWidth="1"
   301→      />
   302→
   303→      {/* Terrace steps */}
   304→      {layers.map((layer, i) => {
   305→        const x1 = i * stepW * 0.7 + stepW * 0.3;
   306→        const y1 = h * 0.1 + i * stepH * 0.7;
   307→        const x2 = x1 + stepW * 1.2;
   308→        const y2 = y1;
   309→        const x3 = x2;
   310→        const y3 = y1 + stepH * 0.7;
   311→        const x4 = x1;
   312→        const y4 = y3;
   313→
   314→        return (
   315→          <g key={layer.id || i}>
   316→            {/* Terrace platform */}
   317→            <path
   318→              d={`M ${x1},${y1} L ${x2},${y2} L ${x3},${y3} L ${x4},${y4} Z`}
   319→              fill={TERRACE_COLORS[i % TERRACE_COLORS.length]}
   320→              stroke="#065f46"
   321→              strokeWidth="1.5"
   322→              opacity="0.85"
   323→            />
   324→            {/* Sediment layer */}
   325→            {layer.sedimentDepthCm > 0 && (
   326→              <rect
   327→                x={x1 + 4}
   328→                y={y2 - Math.min(layer.sedimentDepthCm * 0.8, stepH * 0.3)}
   329→                width={(x2 - x1) - 8}
   330→                height={Math.min(layer.sedimentDepthCm * 0.8, stepH * 0.3)}
   331→                fill="url(#sedimentGrad)"
   332→                rx="2"
   333→              />
   334→            )}
   335→            {/* Water flow arrow */}
   336→            {i < layerCount - 1 && (
   337→              <>
   338→                <line
   339→                  x1={x2 - stepW * 0.3}
   340→                  y1={y2 + 4}
   341→                  x2={x2 - stepW * 0.1}
   342→                  y2={y3 + stepH * 0.3}
   343→                  stroke="#3b82f6"
   344→                  strokeWidth="2"
   345→                  strokeDasharray="4,2"
   346→                  opacity="0.7"
   347→                />
   348→                <polygon
   349→                  points={`${x2 - stepW * 0.1},${y3 + stepH * 0.3} ${x2 - stepW * 0.18},${y3 + stepH * 0.15} ${x2 - stepW * 0.05},${y3 + stepH * 0.2}`}
   350→                  fill="#3b82f6"
   351→                  opacity="0.7"
   352→                />
   353→              </>
   354→            )}
   355→            {/* Check dam indicator */}
   356→            {checkDams.filter(d => d.terraceLayerId === layer.id).length > 0 && (
   357→              <rect
   358→                x={x2 - 6}
   359→                y={y2}
   360→                width={8}
   361→                height={y3 - y2}
   362→                fill="#f59e0b"
   363→                stroke="#b45309"
   364→                strokeWidth="1"
   365→                rx="1"
   366→              />
   367→            )}
   368→            {/* Layer label */}
   369→            <text
   370→              x={x1 + (x2 - x1) / 2}
   371→              y={y1 + (y3 - y1) / 2 + 4}
   372→              textAnchor="middle"
   373→              fill="white"
   374→              fontSize={compact ? 9 : 11}
   375→              fontWeight="bold"
   376→            >
   377→              ชั้น {layer.layerNumber || i + 1}
   378→            </text>
   379→            {!compact && (
   380→              <text
   381→                x={x1 + (x2 - x1) / 2}
   382→                y={y1 + (y3 - y1) / 2 + 16}
   383→                textAnchor="middle"
   384→                fill="#a7f3d0"
   385→                fontSize={8}
   386→              >
   387→                {layer.cropType || 'พืช'} · {layer.sedimentDepthCm || 0} cm
   388→              </text>
   389→            )}
   390→          </g>
   391→        );
   392→      })}
   393→
   394→      {/* Legend */}
   395→      {!compact && (
   396→        <g transform={`translate(10, ${h - 30})`}>
   397→          <rect x="0" y="0" width="12" height="8" fill="url(#sedimentGrad)" rx="1" />
   398→          <text x="16" y="7" fill="#94a3b8" fontSize="8">ตะกอน</text>
   399→          <rect x="55" y="0" width="6" height="10" fill="#f59e0b" stroke="#b45309" rx="1" />
   400→          <text x="66" y="7" fill="#94a3b8" fontSize="8">ฝาย</text>
   401→          <line x1="100" y1="5" x2="120" y2="5" stroke="#3b82f6" strokeWidth="1.5" strokeDasharray="3,2" />
   402→          <text x="124" y="7" fill="#94a3b8" fontSize="8">น้ำไหล</text>
   403→        </g>
   404→      )}
   405→    </svg>
   406→  );
   407→}

   797→          plotId: calcPlotId,
   798→          monitoringPeriodStart: calcStartDate || undefined,
   799→          monitoringPeriodEnd: calcEndDate || undefined,
   800→          includeDetailReport: true,
   801→        }),
   802→      });
   803→      const json = await res.json();
   804→      if (json.success) {
   805→        const data = json.data;
   806→        // Enrich with sediment calculation data
   807→        const plot = plots.find(p => p.plotId === calcPlotId);
   808→        const plotTerraceData = terraces.filter(t => t.plotId === calcPlotId);
   809→        const plotDamData = checkDams.filter(d => d.plotId === calcPlotId);
   810→
   811→        const deltaSOC = (plot?.projectSOC || 30) - (plot?.baselineSOC || 25);
   812→        const areaHa = plot?.areaHa || 4;
   813→        const monitoringYears = data?.monitoringPeriod?.years || 1;
   814→
   815→        // 1. SOC Stock Change
   816→        const socStockChange = deltaSOC * (44 / 12) * areaHa * monitoringYears;
   817→
   818→        // 2. Sediment Trapped
   819→        const totalSediment = plotTerraceData.length > 0
   820→          ? plotTerraceData.reduce((s, t) => s + (t.sedimentTrappedKg || 0), 0) / 1000
   821→          : (plot?.baselineSediment || 8) - (plot?.projectSediment || 3);
   822→        const avgSocPct = plotTerraceData.length > 0
   823→          ? plotTerraceData.reduce((s, t) => s + (t.socPct || 1.5), 0) / plotTerraceData.length / 100
   824→          : 0.02;
   825→        const sedimentTrappedCredit = totalSediment * avgSocPct * (44 / 12);

   827→        // 3. Check Dam Carbon
   828→        const totalDamSediment = plotDamData.length > 0
   829→          ? plotDamData.reduce((s, d) => s + (d.sedimentTrappedKg || 0), 0) / 1000
   830→          : totalSediment * 0.65;

   832→

   835→

   869→            unit: 'tCO2e',
   870→            byLayer: plotTerraceData.length > 0
   871→              ? plotTerraceData.map(t => ({
   872→                  layer: t.layerNumber,
   873→                  sedimentKg: t.sedimentTrappedKg || 0,
   874→                  socPct: t.socPct || 1.5,
   875→                  credit: +((t.sedimentTrappedKg / 1000) * (t.socPct / 100) * (44 / 12)).toFixed(3),

   882→                })),

   890→            byDam: plotDamData.length > 0
   891→              ? plotDamData.map(d => ({
