   267→   267→interface SidebarContentProps {
   268→   268→  collapsed?: boolean;
   269→   269→  activeModule: ModuleKey;
   270→   270→  onModuleChange: (key: ModuleKey) => void;
   271→   271→  menuItemsWithBadges?: MenuItem[];

   273→   273→
   274→   274→function SidebarContent({ collapsed = false, activeModule, onModuleChange, menuItemsWithBadges }: SidebarContentProps) {
   275→   275→  const items = menuItemsWithBadges || menuItems;
   276→   276→  const scrollRef = React.useRef<HTMLDivElement>(null);
   277→   277→  const [canScrollDown, setCanScrollDown] = useState(false);
   278→   278→  const [canScrollUp, setCanScrollUp] = useState(false);
   279→   279→
   280→   280→  // Pre-compute which items should show section headers
   281→   281→  const itemsWithSections = items.map((item, idx) => {
   282→   282→    const prevItem = idx > 0 ? items[idx - 1] : null;
   283→   283→    const showSection = item.section && (!prevItem || prevItem.section !== item.section);
   284→   284→    return { ...item, showSection };
   285→   285→  });
   286→   286→
   287→   287→  const checkScroll = React.useCallback(() => {
   288→   288→    const el = scrollRef.current;
   289→   289→    if (el) {
   290→   290→      setCanScrollUp(el.scrollTop > 4);
   291→   291→      setCanScrollDown(el.scrollTop + el.clientHeight < el.scrollHeight - 4);
   292→   292→    }
   293→   293→  }, []);
   294→   294→
   295→   295→  useEffect(() => {
   296→   296→    checkScroll();
   297→   297→    window.addEventListener('resize', checkScroll);
   298→   298→    return () => window.removeEventListener('resize', checkScroll);
   299→   299→  }, [checkScroll, collapsed]);
   300→   300→
   301→   301→  // Auto-scroll to active item when it changes
   302→   302→  useEffect(() => {
   303→   303→    const timer = setTimeout(() => {
   304→   304→      const el = scrollRef.current;
   305→   305→      if (!el) return;
   306→   306→      const activeBtn = el.querySelector('[data-active="true"]');
   307→   307→      if (activeBtn) {
   308→   308→        activeBtn.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
   309→   309→      }
   310→   310→      checkScroll();
   311→   311→    }, 100);
   312→   312→    return () => clearTimeout(timer);
   313→   313→  }, [activeModule, checkScroll]);
   314→   314→
   315→   315→  return (
   316→   316→    <div className="flex h-full flex-col bg-slate-900 text-white">
   317→   317→      {/* Logo Area */}
   318→   318→      <div className="flex items-center gap-3 px-4 py-5 border-b border-slate-700/50">
   319→   319→        <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-emerald-600 text-white font-bold text-sm">
   320→   320→          dM
   321→   321→        </div>
   322→   322→        {!collapsed && (
   323→   323→          <div className="flex flex-col overflow-hidden">
   324→   324→            <span className="text-sm font-bold tracking-tight">dMRV Platform</span>
   325→   325→            <span className="text-[11px] text-slate-400">Enterprise v4.0</span>
   326→   326→          </div>
   327→   327→        )}
   328→   328→      </div>
   329→   329→
   330→   330→      {/* Scroll indicator top */}
   331→   331→      {canScrollUp && (
   332→   332→        <div className="relative h-0 z-10">
   333→   333→          <div className="absolute inset-x-0 top-0 h-6 bg-gradient-to-b from-slate-900 to-transparent pointer-events-none" />
   334→   334→        </div>
   335→   335→      )}
   336→   336→
   337→   337→      {/* Navigation */}
   338→   338→      <div
   339→   339→        ref={scrollRef}
   340→   340→        onScroll={checkScroll}
   341→   341→        className="flex-1 px-2 py-3 overflow-y-auto sidebar-scroll"
   342→   342→      >
   343→   343→        <nav className="flex flex-col gap-0.5">
   344→   344→          {itemsWithSections.map((item) => {
   345→   345→            const isActive = activeModule === item.key;
   346→   346→            const Icon = item.icon;
   347→   347→
   348→   348→            return (
   349→   349→              <React.Fragment key={item.key}>
   350→   350→                {item.showSection && !collapsed && (
   351→   351→                  <div className="pt-3 pb-1 px-3">
   352→   352→                    <span className="text-[9px] font-bold text-slate-500 uppercase tracking-widest">{item.section}</span>
   353→   353→                  </div>
   354→   354→                )}
   355→   355→                {item.showSection && collapsed && (
   356→   356→                  <div className="pt-2 pb-1 mx-2">
   357→   357→                    <div className="border-t border-slate-700/50" />
   358→   358→                  </div>
   359→   359→                )}
   360→   360→                <button
   361→   361→                  data-active={isActive}
   362→   362→                  onClick={() => onModuleChange(item.key)}
   363→   363→                  className={`flex items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition-all duration-150 ${
   364→   364→                    isActive
   365→   365→                      ? 'bg-emerald-600/20 text-emerald-400 shadow-sm'
   366→   366→                      : 'text-slate-300 hover:bg-slate-800 hover:text-white'
   367→   367→                  } ${collapsed ? 'justify-center' : ''}`}
   368→   368→                  title={collapsed ? item.label : undefined}
   369→   369→                >
   370→   370→                  <Icon className={`h-4 w-4 shrink-0 ${isActive ? 'text-emerald-400' : ''}`} />
   371→   371→                  {!collapsed && (
   372→   372→                    <>
   373→   373→                      <span className="truncate">{item.label}</span>
   374→   374→                      {item.badge && (
   375→   375→                        <Badge className={`ml-auto h-5 min-w-[20px] justify-center text-[10px] text-white ${item.badgeColor || 'bg-emerald-600 hover:bg-emerald-600'}`}>
   376→   376→                          {item.badge}
   377→   377→                        </Badge>
   378→   378→                      )}
   379→   379→                    </>
   380→   380→                  )}
   381→   381→                  {isActive && !collapsed && !item.badge && (
   382→   382→                    <div className="ml-auto h-1.5 w-1.5 rounded-full bg-emerald-400" />
   383→   383→                  )}
   384→   384→                </button>
   385→   385→              </React.Fragment>
   386→   386→            );
   387→   387→          })}
   388→   388→        </nav>
   389→   389→      </div>
   390→   390→
   391→   391→      {/* Scroll indicator bottom */}
   392→   392→      {canScrollDown && !collapsed && (
   393→   393→        <div className="relative h-0 z-10 -mt-6">
   394→   394→          <div className="absolute inset-x-0 bottom-0 h-8 bg-gradient-to-t from-slate-900 via-slate-900/90 to-transparent pointer-events-none flex items-end justify-center pb-1">
   395→   395→            <span className="text-[9px] text-slate-500 flex items-center gap-1">
   396→   396→              <ChevronLeft className="h-3 w-3 rotate-90" />
   397→   397→              Scroll for more
   398→   398→              <ChevronLeft className="h-3 w-3 -rotate-90" />
   399→   399→            </span>
   400→   400→          </div>
   401→   401→        </div>
   402→   402→      )}
   403→   403→
   404→   404→      {/* Footer */}
   405→   405→      {!collapsed && (
   406→   406→        <div className="border-t border-slate-700/50 px-4 py-3">
   407→   407→          <div className="flex items-center gap-2 text-xs text-slate-500">
   408→   408→            <CircleDot className="h-3 w-3 text-emerald-500" />
   409→   409→            <span>System Online</span>
   410→   410→            <span className="ml-auto">v4.0.0</span>
   411→   411→          </div>
   412→   412→        </div>
   413→   413→      )}
   414→   414→    </div>
   415→   415→  );
   416→   416→}
