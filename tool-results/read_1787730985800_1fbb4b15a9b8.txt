    77→export default function CarbonView() {
    78→  const { toast } = useToast();
    79→  const [results, setResults] = useState<CalcResult[]>([]);
    80→  const [credits, setCredits] = useState<Credit[]>([]);
    81→  const [summary, setSummary] = useState<CarbonSummary | null>(null);
    82→  const [projects, setProjects] = useState<Project[]>([]);
    83→  const [loading, setLoading] = useState(true);
    84→  const [calculating, setCalculating] = useState(false);
    85→  const [selectedProjectId, setSelectedProjectId] = useState('');
    86→  const [createCredit, setCreateCredit] = useState(false);
    87→
    88→  // Biochar state
    89→  const [biocharInput, setBiocharInput] = useState({ biocharProduced: '', stabilityFactor: '0.85' });
    90→  // AWD state
    91→  const [awdInput, setAwdInput] = useState({ area: '', baseline: '3.5', mitigationFactor: '0.5' });
    92→  // Biogas state
    93→  const [biogasInput, setBiogasInput] = useState({ ch4Captured: '', fuelDisplacement: '' });
    94→  // Solar state
    95→  const [solarInput, setSolarInput] = useState({ generation: '', gridFactor: '0.5086' });
    96→  // Blue Carbon state
    97→  const [blueInput, setBlueInput] = useState({ area: '', years: '' });
    98→
    99→  const fetchCredits = useCallback(async () => {
   100→    try {
   101→      const res = await fetch('/api/dmrv/carbon');
   102→      if (!res.ok) throw new Error('Failed to fetch credits');
   103→      const data = await res.json();
   104→      setCredits(data.credits || []);
   105→      setSummary(data.summary || null);
   106→    } catch {
   107→      toast({ title: 'Error', description: 'Failed to load carbon credits', variant: 'destructive' });
   108→    }
   109→  }, [toast]);
   110→
   111→  const fetchProjects = useCallback(async () => {
   112→    try {
   113→      const res = await fetch('/api/dmrv');
   114→      if (!res.ok) throw new Error('Failed to fetch projects');
   115→      const data = await res.json();
   116→      setProjects(data.projects || []);
   117→    } catch {
   118→      toast({ title: 'Error', description: 'Failed to load projects', variant: 'destructive' });
   119→    }
   120→  }, [toast]);
   121→
   122→  useEffect(() => {
   123→    const init = async () => {
   124→      setLoading(true);
   125→      await Promise.all([fetchCredits(), fetchProjects()]);
   126→      setLoading(false);
   127→    };
   128→    init();
   129→  }, [fetchCredits, fetchProjects]);
   130→
   131→  const handleRefresh = async () => {
   132→    setLoading(true);
   133→    await fetchCredits();
   134→    setLoading(false);
   135→    toast({ title: 'Refreshed', description: 'Carbon credits data updated' });
   136→  };
   137→
   138→  const calculateBiochar = async () => {
   139→    const produced = parseFloat(biocharInput.biocharProduced);
   140→    if (isNaN(produced) || produced <= 0) {
   141→      toast({ title: 'Invalid Input', description: 'Biochar produced must be a positive number', variant: 'destructive' });
   142→      return;
   143→    }
   144→    setCalculating(true);
   145→    try {
   146→      const res = await fetch('/api/dmrv/carbon', {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify({
   150→          trackType: 'biochar',
   151→          methodology: 'IPCC-2023',
   152→          mass: produced,
   153→          carbonFraction: 0.7,
   154→          stability: parseFloat(biocharInput.stabilityFactor),
   155→          projectId: selectedProjectId || undefined,
   156→          createCredit: createCredit && !!selectedProjectId,
   157→        }),
   158→      });
   159→      if (!res.ok) throw new Error('Calculation failed');
   160→      const data = await res.json();
   161→      const calc = data.calculation;
   162→      setResults((prev) => [...prev, { type: 'biochar', tco2e: calc.tco2e, methodology: 'IPCC 2019 — Biochar Carbon Stability', formula: calc.formula, breakdown: calc.breakdown }]);
   163→      toast({ title: 'Calculation Complete', description: `Biochar: ${calc.tco2e.toFixed(2)} tCO2e${data.credit ? ' (Credit created)' : ''}` });
   164→      if (data.credit) await fetchCredits();
   165→    } catch {
   166→      toast({ title: 'Error', description: 'Carbon calculation failed', variant: 'destructive' });
   167→    } finally {
   168→      setCalculating(false);
   169→    }
   170→  };
   171→
   172→  const calculateAWD = async () => {
   173→    const area = parseFloat(awdInput.area);
   174→    if (isNaN(area) || area <= 0) {
   175→      toast({ title: 'Invalid Input', description: 'Area must be a positive number', variant: 'destructive' });
   176→      return;
   177→    }
   178→    setCalculating(true);
   179→    try {
   180→      const res = await fetch('/api/dmrv/carbon', {
   181→        method: 'POST',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify({
   184→          trackType: 'awd',
   185→          methodology: 'IPCC-2023',
   186→          area,
   187→          baseline: parseFloat(awdInput.baseline),
   188→          mitigationFactor: parseFloat(awdInput.mitigationFactor),
   189→          projectId: selectedProjectId || undefined,
   190→          createCredit: createCredit && !!selectedProjectId,
   191→        }),
   192→      });
   193→      if (!res.ok) throw new Error('Calculation failed');
   194→      const data = await res.json();
   195→      const calc = data.calculation;
   196→      setResults((prev) => [...prev, { type: 'awd', tco2e: calc.tco2e, methodology: 'IRRI 2023 — AWD Methane Reduction', formula: calc.formula, breakdown: calc.breakdown }]);
   197→      toast({ title: 'Calculation Complete', description: `AWD: ${calc.tco2e.toFixed(2)} tCO2e${data.credit ? ' (Credit created)' : ''}` });
   198→      if (data.credit) await fetchCredits();
   199→    } catch {
   200→      toast({ title: 'Error', description: 'Carbon calculation failed', variant: 'destructive' });
   201→    } finally {
   202→      setCalculating(false);
   203→    }
   204→  };
   205→
   206→  const calculateBiogas = async () => {
   207→    const ch4 = parseFloat(biogasInput.ch4Captured);
   208→    if (isNaN(ch4) || ch4 <= 0) {
   209→      toast({ title: 'Invalid Input', description: 'CH4 captured must be a positive number', variant: 'destructive' });
   210→      return;
   211→    }
   212→    setCalculating(true);
   213→    try {
   214→      const res = await fetch('/api/dmrv/carbon', {
   215→        method: 'POST',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({
   218→          trackType: 'biogas',
   219→          methodology: 'IPCC-2023',
   220→          ch4Capture: ch4,
   221→          fuelDisplacement: parseFloat(biogasInput.fuelDisplacement) || 0,
   222→          projectId: selectedProjectId || undefined,
   223→          createCredit: createCredit && !!selectedProjectId,
   224→        }),
   225→      });
   226→      if (!res.ok) throw new Error('Calculation failed');
   227→      const data = await res.json();
   228→      const calc = data.calculation;
   229→      setResults((prev) => [...prev, { type: 'biogas', tco2e: calc.tco2e, methodology: 'IPCC AR5 — Biogas Methane Capture', formula: calc.formula, breakdown: calc.breakdown }]);
   230→      toast({ title: 'Calculation Complete', description: `Biogas: ${calc.tco2e.toFixed(2)} tCO2e${data.credit ? ' (Credit created)' : ''}` });
   231→      if (data.credit) await fetchCredits();
   232→    } catch {
   233→      toast({ title: 'Error', description: 'Carbon calculation failed', variant: 'destructive' });
   234→    } finally {
   235→      setCalculating(false);
   236→    }
   237→  };
   238→
   239→  const calculateSolar = async () => {
   240→    const gen = parseFloat(solarInput.generation);
   241→    if (isNaN(gen) || gen <= 0) {
   242→      toast({ title: 'Invalid Input', description: 'Generation must be a positive number', variant: 'destructive' });
   243→      return;
   244→    }
   245→    setCalculating(true);
   246→    try {
   247→      const res = await fetch('/api/dmrv/carbon', {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({
   251→          trackType: 'solar',
   252→          methodology: 'IPCC-2023',
   253→          kwh: gen * 1000, // Convert MWh to kWh
   254→          gridEF: parseFloat(solarInput.gridFactor),
   255→          projectId: selectedProjectId || undefined,
   256→          createCredit: createCredit && !!selectedProjectId,
   257→        }),
   258→      });
   259→      if (!res.ok) throw new Error('Calculation failed');
   260→      const data = await res.json();
   261→      const calc = data.calculation;

   263→      toast({ title: 'Calculation Complete', description: `Solar: ${calc.tco2e.toFixed(2)} tCO2e${data.credit ? ' (Credit created)' : ''}` });

   265→    } catch {
   266→      toast({ title: 'Error', description: 'Carbon calculation failed', variant: 'destructive' });
   267→    } finally {
   268→      setCalculating(false);
   269→    }
   270→  };
   271→
   272→  const calculateBlue = async () => {
   273→    const area = parseFloat(blueInput.area);
   274→    const years = parseFloat(blueInput.years);
   275→    if (isNaN(area) || isNaN(years) || area <= 0 || years <= 0) {
   276→      toast({ title: 'Invalid Input', description: 'Area and years must be positive numbers', variant: 'destructive' });
   277→      return;
   278→    }
   279→    // Blue carbon uses area-based formula, map to AWD-like endpoint with appropriate values
   280→    setCalculating(true);
   281→    try {
   282→      const res = await fetch('/api/dmrv/carbon', {
   283→        method: 'POST',
   284→        headers: { 'Content-Type': 'application/json' },
   285→        body: JSON.stringify({
   286→          trackType: 'awd',
   287→          methodology: 'T-VER-FOREST',
   288→          baseline: 9.9, // Blue carbon sequestration rate
   289→          mitigationFactor: 1.0,
   290→          area: area * years,
   291→          projectId: selectedProjectId || undefined,
   292→          createCredit: createCredit && !!selectedProjectId,
   293→        }),
   294→      });
   295→      if (!res.ok) throw new Error('Calculation failed');
   296→      const data = await res.json();
   297→      const calc = data.calculation;
   298→      setResults((prev) => [...prev, { type: 'bluecarbon', tco2e: calc.tco2e, methodology: 'Murray et al. 2011 — Blue Carbon Sequestration', formula: calc.formula, breakdown: calc.breakdown }]);
   299→      toast({ title: 'Calculation Complete', description: `Blue Carbon: ${calc.tco2e.toFixed(2)} tCO2e${data.credit ? ' (Credit created)' : ''}` });
   300→      if (data.credit) await fetchCredits();
   301→    } catch {
   302→      toast({ title: 'Error', description: 'Carbon calculation failed', variant: 'destructive' });
   303→    } finally {
   304→      setCalculating(false);
   305→    }
   306→  };
   307→
   308→  if (loading) {
   309→    return (
   310→      <div className="space-y-6">
   311→        <div>
   312→          <Skeleton className="h-8 w-64" />
   313→          <Skeleton className="h-4 w-96 mt-2" />
   314→        </div>
   315→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   316→          {[1, 2, 3, 4].map((i) => (
   317→            <Card key={i}>
   318→              <CardContent className="p-4 text-center">
   319→                <Skeleton className="h-8 w-16 mx-auto" />
   320→                <Skeleton className="h-3 w-20 mx-auto mt-2" />
   321→              </CardContent>
   322→            </Card>
   323→          ))}
   324→        </div>
   325→        <div className="grid grid-cols-1 gap-6 lg:grid-cols-3">
   326→          <div className="lg:col-span-2"><Card><CardContent className="p-6"><Skeleton className="h-64" /></CardContent></Card></div>
   327→          <Card><CardContent className="p-6"><Skeleton className="h-64" /></CardContent></Card>
   328→        </div>
   329→      </div>
   330→    );
   331→  }
   332→
   333→  return (
   334→    <div className="space-y-6">
   335→      {/* Page Title */}
   336→      <div className="flex items-center justify-between">
   337→        <div>
   338→          <h2 className="text-2xl font-bold tracking-tight text-gray-900">Carbon Quantification</h2>
   339→          <p className="text-muted-foreground text-sm mt-1">
   340→            Calculate emission reductions (tCO2e) using approved methodologies
   341→          </p>
   342→        </div>
   343→        <Button variant="outline" size="sm" onClick={handleRefresh} className="gap-2">
   344→          <RefreshCw className="h-3.5 w-3.5" />
   345→          Refresh
   346→        </Button>
   347→      </div>
   348→
   349→      {/* Summary Cards */}
   350→      {summary && (
   351→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   352→          <Card>
   353→            <CardContent className="p-4 text-center">
   354→              <p className="text-2xl font-bold text-emerald-600">{summary.totalCredits.toLocaleString('en-US', { maximumFractionDigits: 1 })}</p>
   355→              <p className="text-xs text-muted-foreground">Total tCO2e</p>
   356→            </CardContent>
   357→          </Card>
   358→          <Card>
   359→            <CardContent className="p-4 text-center">
   360→              <p className="text-2xl font-bold text-green-600">{summary.available.toLocaleString('en-US', { maximumFractionDigits: 1 })}</p>
   361→              <p className="text-xs text-muted-foreground">Available</p>
   362→            </CardContent>
   363→          </Card>
   364→          <Card>
   365→            <CardContent className="p-4 text-center">
   366→              <p className="text-2xl font-bold text-teal-600">{summary.traded.toLocaleString('en-US', { maximumFractionDigits: 1 })}</p>
   367→              <p className="text-xs text-muted-foreground">Traded</p>
   368→            </CardContent>
   369→          </Card>
   370→          <Card>
   371→            <CardContent className="p-4 text-center">
   372→              <p className="text-2xl font-bold text-slate-600">{summary.retired.toLocaleString('en-US', { maximumFractionDigits: 1 })}</p>
   373→              <p className="text-xs text-muted-foreground">Retired</p>
   374→            </CardContent>
   375→          </Card>
   376→        </div>
   377→      )}
   378→
   379→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-3">
   380→        {/* Calculator Tabs */}
   381→        <div className="lg:col-span-2">
   382→          <Card>
   383→            <CardHeader>
   384→              <CardTitle className="text-base flex items-center gap-2">
   385→                <Calculator className="h-4 w-4 text-emerald-600" />
   386→                Carbon Calculator
   387→              </CardTitle>
   388→            </CardHeader>
   389→            <CardContent>
   390→              {/* Project Selector and Create Credit */}
   391→              <div className="flex flex-col sm:flex-row gap-4 mb-6 p-4 bg-slate-50 rounded-lg">
   392→                <div className="flex-1 space-y-2">
   393→                  <Label>Assign to Project</Label>
   394→                  <Select value={selectedProjectId} onValueChange={setSelectedProjectId}>
   395→                    <SelectTrigger>
   396→                      <SelectValue placeholder="Select a project..." />
   397→                    </SelectTrigger>
   398→                    <SelectContent>
   399→                      {projects.map((p) => (
   400→                        <SelectItem key={p.id} value={p.id}>
   401→                          {p.name} ({p.methodology})
   402→                        </SelectItem>
   403→                      ))}
   404→                    </SelectContent>
   405→                  </Select>
   406→                </div>
   407→                <div className="flex items-center gap-2 pt-6">
   408→                  <Checkbox
   409→                    id="createCredit"
   410→                    checked={createCredit}
   411→                    onCheckedChange={(checked) => setCreateCredit(checked === true)}
   412→                    disabled={!selectedProjectId}
   413→                  />
   414→                  <Label htmlFor="createCredit" className={`text-sm ${!selectedProjectId ? 'text-muted-foreground' : ''}`}>
   415→                    Create Credit
   416→                  </Label>
   417→                </div>
   418→              </div>
   419→
   420→              <Tabs defaultValue="biochar">
   421→                <TabsList className="flex-wrap h-auto gap-1">
   422→                  <TabsTrigger value="biochar">Biochar</TabsTrigger>
   423→                  <TabsTrigger value="awd">AWD</TabsTrigger>
   424→                  <TabsTrigger value="biogas">Biogas</TabsTrigger>
   425→                  <TabsTrigger value="solar">Solar</TabsTrigger>
   426→                  <TabsTrigger value="bluecarbon">Blue Carbon</TabsTrigger>

   428→
   429→                <TabsContent value="biochar" className="space-y-4 mt-4">

   439→                      />
   440→                    </div>
   441→                    <div className="space-y-2">
   442→                      <Label>Carbon Stability Factor</Label>
   443→                      <Input
   444→                        type="number"

   446→                        value={biocharInput.stabilityFactor}
   447→                        onChange={(e) => setBiocharInput({ ...biocharInput, stabilityFactor: e.target.value })}
   448→                      />
   449→                    </div>
   450→                  </div>
   451→                  <Button onClick={calculateBiochar} disabled={calculating} className="bg-emerald-600 hover:bg-emerald-700 gap-2">
   452→                    <Leaf className="h-4 w-4" />
   453→                    {calculating ? 'Calculating...' : 'Calculate'}
   454→                  </Button>
   455→                </TabsContent>
   456→
   457→                <TabsContent value="awd" className="space-y-4 mt-4">
   458→                  <p className="text-sm text-muted-foreground">Calculate methane reduction from Alternate Wetting and Drying (AWD) in rice cultivation.</p>
   459→                  <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   460→                    <div className="space-y-2">

   485→                      />
   486→                    </div>
   487→                  </div>
   488→                  <Button onClick={calculateAWD} disabled={calculating} className="bg-emerald-600 hover:bg-emerald-700 gap-2">
   489→                    <Leaf className="h-4 w-4" />
   490→                    {calculating ? 'Calculating...' : 'Calculate'}
   491→                  </Button>
   492→                </TabsContent>
   493→
