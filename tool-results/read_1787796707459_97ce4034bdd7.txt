    22→type CreditStatus = 'Available' | 'Traded' | 'Retired';
    23→
    24→interface MarketplaceCredit {
    25→  id: string;
    26→  tokenId: string;
    27→  amount: number;
    28→  project: string;
    29→  methodology: string;
    30→  location: string;
    31→  metadata: Record<string, unknown>;
    32→}
    33→
    34→interface TradedCredit {
    35→  id: string;
    36→  tokenId: string;
    37→  amount: number;
    38→  buyer: string | null;
    39→  project: string;
    40→  tradedAt: string;
    41→}
    42→
    43→interface RetiredCredit {
    44→  id: string;
    45→  tokenId: string;
    46→  amount: number;
    47→  buyer: string | null;
    48→  project: string;
    49→  retiredAt: string;
    50→}
    51→
    52→interface MarketplaceSummary {
    53→  totalCredits: number;
    54→  availableVolume: number;
    55→  tradedVolume: number;
    56→  retiredVolume: number;
    57→  availableCount: number;
    58→  tradedCount: number;
    59→  retiredCount: number;
    60→}
    61→
    62→interface Project {
    63→  id: string;
    64→  name: string;
    65→}

    85→const methodologyColors: Record<string, string> = {
    86→  'IPCC-2023': 'bg-violet-100 text-violet-700',

    91→
    92→function getMethodologyClass(m: string) {
    93→  return methodologyColors[m] || 'bg-gray-100 text-gray-600';
    94→}
    95→
    96→function truncateTokenId(id: string) {
    97→  if (id.length <= 12) return id;
    98→  return `${id.slice(0, 8)}…${id.slice(-4)}`;
    99→}

   123→
   124→export default function MarketplaceView() {
   125→  const { toast } = useToast();
   126→  const [loading, setLoading] = useState(true);
   127→  const [available, setAvailable] = useState<MarketplaceCredit[]>([]);
   128→  const [traded, setTraded] = useState<TradedCredit[]>([]);
   129→  const [retired, setRetired] = useState<RetiredCredit[]>([]);
   130→  const [summary, setSummary] = useState<MarketplaceSummary | null>(null);
   131→  const [projects, setProjects] = useState<Project[]>([]);
   132→
   133→  // Mint form
   134→  const [mintProjectId, setMintProjectId] = useState('');
   135→  const [mintTrackType, setMintTrackType] = useState('biochar');
   136→  const [mintAmount, setMintAmount] = useState('');
   137→  const [minting, setMinting] = useState(false);
   138→
   139→  // Trade form
   140→  const [tradeId, setTradeId] = useState('');
   141→  const [tradeBuyer, setTradeBuyer] = useState('');
   142→  const [trading, setTrading] = useState(false);
   143→
   144→  // Retire form
   145→  const [retireId, setRetireId] = useState('');
   146→  const [retiring, setRetiring] = useState(false);
   147→
   148→  const fetchData = useCallback(async () => {
   149→    try {
   150→      const [marketRes, projectsRes] = await Promise.all([
   151→        fetch('/api/dmrv/marketplace'),
   152→        fetch('/api/dmrv'),
   153→      ]);
   154→      if (!marketRes.ok || !projectsRes.ok) throw new Error('Failed to fetch data');
   155→
   156→      const marketData = await marketRes.json();
   157→      const projectsData = await projectsRes.json();
   158→
   159→      setAvailable(marketData.marketplace?.available || []);
   160→      setTraded(marketData.marketplace?.traded || []);
   161→      setRetired(marketData.marketplace?.retired || []);
   162→      setSummary(marketData.summary || null);
   163→      setProjects((projectsData.projects || []).map((p: { id: string; name: string }) => ({ id: p.id, name: p.name })));
   164→    } catch (err) {
   165→      toast({ title: 'Error', description: 'Failed to load marketplace data', variant: 'destructive' });
   166→    } finally {
   167→      setLoading(false);
   168→    }
   169→  }, [toast]);
   170→
   171→  useEffect(() => {
   172→    fetchData();
   173→  }, [fetchData]);
   174→
   175→  const handleMint = async () => {
   176→    const amount = parseFloat(mintAmount);
   177→    if (!mintProjectId || isNaN(amount) || amount <= 0) {
   178→      toast({ title: 'Validation Error', description: 'Project and valid amount are required', variant: 'destructive' });
   179→      return;
   180→    }
   181→    try {
   182→      setMinting(true);
   183→      const res = await fetch('/api/dmrv/marketplace', {
   184→        method: 'POST',
   185→        headers: { 'Content-Type': 'application/json' },
   186→        body: JSON.stringify({ action: 'mint', projectId: mintProjectId, amount, trackType: mintTrackType }),
   187→      });
   188→      if (!res.ok) {
   189→        const errData = await res.json();
   190→        throw new Error(errData.error || 'Mint failed');
   191→      }
   192→      toast({ title: 'Credits Minted', description: `${amount.toLocaleString()} tCO2e minted successfully` });
   193→      setMintAmount('');
   194→      setMintProjectId('');
   195→      await fetchData();
   196→    } catch (err) {
   197→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to mint credits', variant: 'destructive' });
   198→    } finally {
   199→      setMinting(false);
   200→    }
   201→  };
   202→
   203→  const handleTrade = async () => {
   204→    if (!tradeId) {
   205→      toast({ title: 'Validation Error', description: 'Select a credit to trade', variant: 'destructive' });
   206→      return;
   207→    }
   208→    if (!tradeBuyer.trim()) {
   209→      toast({ title: 'Validation Error', description: 'Buyer name is required', variant: 'destructive' });
   210→      return;
   211→    }
   212→    const credit = available.find((c) => c.id === tradeId);
   213→    if (!credit) {
   214→      toast({ title: 'Cannot Trade', description: 'Only available credits can be traded', variant: 'destructive' });
   215→      return;
   216→    }
   217→    try {
   218→      setTrading(true);
   219→      const res = await fetch('/api/dmrv/marketplace', {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify({ action: 'trade', projectId: credit.methodology, creditId: tradeId, buyer: tradeBuyer.trim() }),
   223→      });
   224→      if (!res.ok) {
   225→        const errData = await res.json();
   226→        throw new Error(errData.error || 'Trade failed');
   227→      }
   228→      toast({ title: 'Credit Traded', description: `${credit.tokenId} has been traded to ${tradeBuyer.trim()}` });
   229→      setTradeId('');
   230→      setTradeBuyer('');
   231→      await fetchData();
   232→    } catch (err) {
   233→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to trade credit', variant: 'destructive' });
   234→    } finally {
   235→      setTrading(false);
   236→    }
   237→  };
   238→
   239→  const handleRetire = async () => {
   240→    if (!retireId) {
   241→      toast({ title: 'Validation Error', description: 'Select a credit to retire', variant: 'destructive' });
   242→      return;
   243→    }
   244→    try {
   245→      setRetiring(true);
   246→      const credit = [...available, ...traded].find((c) => c.id === retireId);
   247→      const projectId = credit ? ('methodology' in credit ? (credit as MarketplaceCredit).methodology : (credit as TradedCredit).project) : '';
   248→      const res = await fetch('/api/dmrv/marketplace', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ action: 'retire', projectId: projectId || 'unknown', creditId: retireId }),
   252→      });
   253→      if (!res.ok) {
   254→        const errData = await res.json();
   255→        throw new Error(errData.error || 'Retire failed');
   256→      }
   257→      toast({ title: 'Credit Retired', description: `Credit has been permanently retired` });
   258→      setRetireId('');
   259→      await fetchData();
   260→    } catch (err) {
   261→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to retire credit', variant: 'destructive' });
   262→    } finally {
   263→      setRetiring(false);
   264→    }
   265→  };
   266→
   267→  const allCredits = [
   268→    ...available.map((c) => ({ ...c, status: 'Available' as CreditStatus, certNumber: c.tokenId, batchId: c.tokenId, projectId: c.project, type: c.methodology, price: (c.metadata?.price as number) || 0, createdAt: '' })),
   269→    ...traded.map((c) => ({ ...c, status: 'Traded' as CreditStatus, certNumber: c.tokenId, batchId: c.tokenId, projectId: c.project, type: '', price: 0, createdAt: c.tradedAt })),
   270→    ...retired.map((c) => ({ ...c, status: 'Retired' as CreditStatus, certNumber: c.tokenId, batchId: c.tokenId, projectId: c.project, type: '', price: 0, createdAt: c.retiredAt })),
   271→  ];
   272→
   273→  const availablePct = useMemo(() => {
   274→    if (!summary || summary.totalCredits === 0) return 0;
   275→    return Math.round((summary.availableVolume / summary.totalCredits) * 100);
   276→  }, [summary]);
   277→
   278→  const tradedPct = useMemo(() => {
   279→    if (!summary || summary.totalCredits === 0) return 0;
   280→    return Math.round((summary.tradedVolume / summary.totalCredits) * 100);
   281→  }, [summary]);
   282→
   283→  const retiredPct = useMemo(() => {
   284→    if (!summary || summary.totalCredits === 0) return 0;
   285→    return Math.round((summary.retiredVolume / summary.totalCredits) * 100);
   286→  }, [summary]);
   287→
   288→  if (loading) {
   289→    return (
   290→      <div className="space-y-6">
   291→        <Skeleton className="h-40 w-full rounded-2xl" />
   292→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   293→          <Skeleton className="h-28 w-full rounded-xl" />
   294→          <Skeleton className="h-28 w-full rounded-xl" />
   295→          <Skeleton className="h-28 w-full rounded-xl" />
   296→          <Skeleton className="h-28 w-full rounded-xl" />
   297→        </div>
   298→        <Skeleton className="h-80 w-full rounded-xl" />
   299→      </div>
   300→    );
   301→  }
   302→
   303→  return (
   304→    <div className="space-y-6">
   305→      {/* ── Hero Banner ──────────────────────────────── */}
   306→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-slate-900 via-emerald-950 to-slate-900 p-6 sm:p-8">
   307→        {/* Decorative blobs */}
   308→        <div className="pointer-events-none absolute -right-20 -top-20 h-72 w-72 rounded-full bg-emerald-500/10 blur-3xl" />
   309→        <div className="pointer-events-none absolute -bottom-16 -left-16 h-56 w-56 rounded-full bg-teal-400/10 blur-3xl" />
   310→
   311→        <div className="relative z-10 flex flex-col lg:flex-row lg:items-center lg:justify-between gap-6">
   312→          {/* Title block */}
   313→          <div className="space-y-3">
   314→            <div className="flex items-center gap-3">
   315→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-500/20">
   316→                <Coins className="h-5 w-5 text-emerald-400" />
   317→              </div>
   318→              <h2 className="text-2xl sm:text-3xl font-bold text-white tracking-tight">Marketplace</h2>
   319→            </div>
   320→            <p className="text-emerald-200/70 text-sm max-w-xl leading-relaxed">
   321→              ตลาดซื้อขายคาร์บอนเครดิตแบบ Tokenized สำหรับ Mint, Trade และ Retire เครดิตคาร์บอนจากโครงการที่ผ่านการรับรองแล้ว รองรับมาตรฐาน T-VER, Verra, Gold Standard
   322→            </p>
   323→            <div className="flex items-center gap-4 text-xs text-emerald-300/60">
   324→              <span className="flex items-center gap-1.5"><CircleDot className="h-3 w-3" />{available.length} Available</span>
   325→              <span className="flex items-center gap-1.5"><ArrowRightLeft className="h-3 w-3" />{traded.length} Traded</span>
   326→              <span className="flex items-center gap-1.5"><ShieldCheck className="h-3 w-3" />{retired.length} Retired</span>
   327→            </div>
   328→          </div>
   329→
   330→          {/* PDF Document Card */}
   331→          <a
   332→            href="/Tokenized-Carbon-Credit-Platform.pdf"
   333→            target="_blank"
   334→            rel="noopener noreferrer"
   335→            className="group flex-shrink-0 w-full lg:w-80 rounded-xl border border-emerald-500/20 bg-white/5 backdrop-blur-sm p-4 hover:bg-white/10 transition-all duration-200 cursor-pointer"
   336→          >
   337→            <div className="flex items-start gap-3">
   338→              <div className="flex h-11 w-11 flex-shrink-0 items-center justify-center rounded-lg bg-red-500/20">
   339→                <FileText className="h-5 w-5 text-red-400" />
   340→              </div>
   341→              <div className="flex-1 min-w-0">
   342→                <p className="text-sm font-semibold text-white truncate">Tokenized Carbon Credit Platform</p>
   343→                <p className="text-xs text-emerald-300/50 mt-0.5">เอกสารอธิบายระบบ Tokenization ครบวงจร 11 บท</p>
   344→                <div className="flex items-center gap-1.5 mt-2 text-xs text-emerald-400 font-medium group-hover:text-emerald-300 transition-colors">
   345→                  <FileDown className="h-3.5 w-3.5" />
   346→                  ดาวน์โหลด PDF (17 หน้า)
   347→                  <ArrowUpRight className="h-3 w-3 ml-auto transition-transform group-hover:translate-x-0.5 group-hover:-translate-y-0.5" />
   348→                </div>
   349→              </div>
   350→            </div>
   351→          </a>
   352→        </div>
   353→      </div>
   354→
   355→      {/* ── Stats Cards ──────────────────────────────── */}
   356→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">

   360→          value={(summary?.availableVolume ?? 0).toLocaleString()}
   361→          color="bg-emerald-600"
   362→          sub={`${summary?.availableCount ?? 0} batches · tCO2e`}
   363→        />
   364→        <StatCard

   370→        />
   371→        <StatCard
   372→          icon={ShieldCheck}
   373→          label="Retired"
   374→          value={(summary?.retiredVolume ?? 0).toLocaleString()}
   375→          color="bg-slate-500"
   376→          sub={`${summary?.retiredCount ?? 0} batches · tCO2e`}

   379→          icon={TrendingUp}
   380→          label="Total Credits"
   381→          value={(summary?.totalCredits ?? 0).toLocaleString()}
   382→          color="bg-teal-600"
   383→          sub="tCO2e minted"
   384→        />
   385→      </div>
   386→
   387→      {/* ── Tabs ──────────────────────────────────────── */}
   388→      <Tabs defaultValue="listing">
   389→        <TabsList className="bg-slate-100">
   390→          <TabsTrigger value="listing" className="gap-1.5 data-[state=active]:bg-white data-[state=active]:shadow-sm">
   391→            <Layers className="h-3.5 w-3.5" />
   392→            <span className="hidden sm:inline">Credit Listing</span>
   393→            <span className="sm:hidden">Listing</span>
   394→          </TabsTrigger>
   395→          <TabsTrigger value="mint" className="gap-1.5 data-[state=active]:bg-white data-[state=active]:shadow-sm">
   396→            <Plus className="h-3.5 w-3.5" />
   397→            <span className="hidden sm:inline">Mint Credits</span>
   398→            <span className="sm:hidden">Mint</span>
   399→          </TabsTrigger>
   400→          <TabsTrigger value="actions" className="gap-1.5 data-[state=active]:bg-white data-[state=active]:shadow-sm">
   401→            <ArrowRightLeft className="h-3.5 w-3.5" />
   402→            <span className="hidden sm:inline">Trade / Retire</span>
   403→            <span className="sm:hidden">Trade</span>
   404→          </TabsTrigger>
   405→          <TabsTrigger value="portfolio" className="gap-1.5 data-[state=active]:bg-white data-[state=active]:shadow-sm">
   406→            <Briefcase className="h-3.5 w-3.5" />
   407→            <span className="hidden sm:inline">Portfolio</span>
   408→            <span className="sm:hidden">Portfolio</span>
   409→          </TabsTrigger>
   410→        </TabsList>
   411→
   412→        {/* ── Tab: Credit Listing ─────────────────────── */}
   413→        <TabsContent value="listing" className="mt-4">
   414→          <Card className="overflow-hidden">
   415→            <CardHeader className="bg-slate-50/80 border-b px-6 py-4">
   416→              <div className="flex items-center justify-between">
   417→                <div>
   418→                  <CardTitle className="text-base flex items-center gap-2">
   419→                    <ShoppingCart className="h-4 w-4 text-emerald-600" />
   420→                    Credit Listing
   421→                  </CardTitle>
   422→                  <CardDescription className="mt-1">รายการเครดิตคาร์บอนทั้งหมดในระบบ Marketplace</CardDescription>
   423→                </div>
   424→                <Button variant="outline" size="sm" onClick={fetchData} className="gap-1.5">

   426→                  Refresh
   427→                </Button>
   428→              </div>
   429→            </CardHeader>
   430→            <CardContent className="p-0">
   431→              <ScrollArea className="max-h-[420px]">
   432→                <Table>
   433→                  <TableHeader>
   434→                    <TableRow className="bg-slate-50/50 hover:bg-slate-50/50">
   435→                      <TableHead className="w-[180px]">Token ID</TableHead>
   436→                      <TableHead>Project</TableHead>
   437→                      <TableHead className="w-[130px]">Methodology</TableHead>
   438→                      <TableHead className="text-right w-[120px]">Amount (tCO2e)</TableHead>
   439→                      <TableHead className="text-center w-[100px]">Status</TableHead>
   440→                    </TableRow>
   441→                  </TableHeader>
   442→                  <TableBody>

   448→                        </TableCell>
   449→                      </TableRow>
   450→                    ) : (

   459→                            </TableCell>
   460→                            <TableCell className="text-sm font-medium">{credit.projectId}</TableCell>
   461→                            <TableCell>
   462→                              {credit.type ? (
   463→                                <Badge variant="outline" className={`text-[10px] h-5 font-semibold ${getMethodologyClass(credit.type)}`}>
   464→                                  {credit.type}
   465→                                </Badge>

   469→                            </TableCell>
   470→                            <TableCell className="text-right font-mono text-sm font-semibold tabular-nums">
